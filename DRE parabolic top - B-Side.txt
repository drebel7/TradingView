iBullishSqueezer = input.bool(defval = false, title = 'bullish', group = 'squeezer', inline = 'squeezer1')

iSqueezerLookback = input.int(defval = 325, title = 'lookback', minval = 100, maxval = 750, step = 5, group = 'squeezer', display = display.none)
iSqueezerMinInitVwapDist = input.float(defval = 11.0, title = 'min vwap init. dist.', minval = 2.0, maxval = 20.0, group = 'squeezer', display = display.none)
iSqueezerMaxVwapDist = input.float(defval = 2.0, title = 'max vwap squeeze dist.', minval = 0.0, maxval = 10.0, group = 'squeezer', display = display.none)

func_sig_bull_squeezer(_atr, _lookback, _gracePeriod, _minInitVwapDist, _maxVwapDist) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localLow = ta.lowest(low, math.max(1, nz(localHighCandlesAgo)))
    localLowCandlesAgo = ta.barssince(low == localLow)
    localHighVwap = ta.vwap(ohlc4, high == localHigh)
    localLowVwap = ta.vwap(ohlc4, low == localLow)
    maDist = localHighVwap - localLowVwap
    sig = localLowCandlesAgo > 50 and localHighVwap[localLowCandlesAgo] > localLow[localLowCandlesAgo] + _minInitVwapDist * _atr and maDist > 0 and maDist < _maxVwapDist * _atr and drelib.crossover(localHighVwap) and localHigh == localHigh[_gracePeriod]
    sig

bullishSqueezerSig = func_sig_bull_squeezer(mainAtr, iStepperLookback, 50, iSqueezerMinInitVwapDist, iSqueezerMaxVwapDist)

alertcondition(gTurnoverCond and bullishSqueezerSig, title = 'DRE Bullish squeezer', message = 'stepper')
