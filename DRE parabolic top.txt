//@version=5
indicator(title = 'DRE parabolic top', overlay = true, max_bars_back = 3000)

iParabolicTop = input.bool(defval=true, title='Parabolic top', group='parabolic top')
iMultipleGain = input.float(defval=1.6, title='Multiple Gain Since Low', minval=1.2, maxval=4, step=0.2, group='parabolic top')
iMultipleInDays = input.int(defval=100, title='Number Of Days To Multiple', minval=5, maxval=150, step=5, group='parabolic top')
iDownDaysAfterTop = input.int(defval=2, title='Down Days After Top', minval=1, maxval=5, group='parabolic top')
iVolDownDaysAfterTop = input.int(defval=2, title='Down Days After Top (Volume)', minval=1, maxval=5, group='parabolic top')

volMa = ta.sma(volume, 30)
closeMa = ta.sma(close, 20)
barsSinceVolDown = ta.barssince(volMa < volMa[3])

topSig = barsSinceVolDown > 15 and close > iMultipleGain * ta.lowest(close, iMultipleInDays) and volume < volume[iVolDownDaysAfterTop] and open < close[iDownDaysAfterTop]

alertcondition(topSig, title='DRE Parabolic TOP sig', message='parabolic TOP')
plotshape(topSig, title='parabolic top', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='top', textcolor=color.new(color.red, 0))


iBullishKicker = input.bool(defval=true, title='Show bullish Kicker', group='kicker')
iBearishKicker = input.bool(defval=true, title='Show bearish Kicker', group='kicker')
iKicker1stCandleSize = input.float(defval=0.5, title='Min size of bearish candle compared to previous candles', minval=0.0, maxval=2.0, step=0.1, group='kicker')
iKicker2ndCandleMinBodySize = input.float(defval=0.5, title='Min body size of bearish candle', minval=0.0, maxval=1.0, step=0.1, group='kicker')

avgCandleSize = ta.sma(high-low, 50)
bodySize = math.abs((open[1]-close[1])/(high[1]-low[1]))
isBullishKicker = open[1]>close[1] and open>open[1] and close>open and volume>0.95*volume[1]
isBaerishKicker = open[1]<close[1] and open<open[1] and close<open and volume>0.95*volume[1]
bullishKickerSig = isBullishKicker and bodySize>iKicker2ndCandleMinBodySize and (high-low)/avgCandleSize>iKicker1stCandleSize and low[1] == ta.lowest(low, 8)
bearishKickerSig = isBaerishKicker and bodySize>iKicker2ndCandleMinBodySize and (high-low)/avgCandleSize>iKicker1stCandleSize and high[1] == ta.highest(high, 8)

alertcondition(bullishKickerSig, title='DRE Bullish Kicker sig', message='bullish kicker')
alertcondition(bearishKickerSig, title='DRE Bearish Kicker sig', message='bearish kicker')
plotshape(bullishKickerSig and iBullishKicker, title='bullish kicker', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='kicker', textcolor=color.new(color.green, 0))
plotshape(bearishKickerSig and iBearishKicker, title='bearish kicker', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='kicker', textcolor=color.new(color.red, 0))


iBullEngulfing = input.bool(defval=true, title='Show bullish Engulfing', group='bullish engulfing')

avgVolume = ta.sma(volume, 10)
bullishEngulfingSig = open[1]>close[1] and open<close and open<close[1] and close>open[1] and low == ta.lowest(low, 8) and volume > avgVolume

alertcondition(bullishEngulfingSig, title='DRE Bullish Engulfing sig', message='bullish engulfing')
plotshape(bullishEngulfingSig and iBullEngulfing, title='bullish engulfing', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='engulfing', textcolor=color.new(color.green, 0))


iBearEngulfing = input.bool(defval=true, title='Show bearish Engulfing', group='bearish engulfing')

bearishEngulfingSig = open[1]<close[1] and open>close and open>close[1] and close<open[1] and high == ta.highest(high, 8) and volume > avgVolume

alertcondition(bearishEngulfingSig, title='DRE Bearish Engulfing sig', message='bearish engulfing')
plotshape(bearishEngulfingSig and iBearEngulfing, title='bearish engulfing', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='engulfing', textcolor=color.new(color.red, 0))


iBearish3Wicks = input.bool(defval=true, title='Show bearish 3 wicks', group='bearish wicks')
avgHighWick = ta.sma(high - math.max(open, close), 3)
avgLowWick = ta.sma(math.min(open, close) - low, 3)
bearish3WicksSig = ta.lowest(high, 3) > math.max(open, close, open[1], close[1], open[2], close[2]) and ta.highest(high, 3) == ta.highest(high, 8) and avgHighWick > 2.5*avgLowWick
alertcondition(bearish3WicksSig, title='DRE Bearish 3 wicks sig', message='bearish wicks')
plotshape(bearish3WicksSig and iBearish3Wicks, title='bearish wicks', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='wicks', textcolor=color.new(color.red, 0))


iBullsh3Wicks = input.bool(defval=true, title='Show bullish 3 wicks', group='bullish wicks')
bullish3WicksSig = ta.highest(low, 3) < math.min(open, close, open[1], close[1], open[2], close[2]) and ta.lowest(low, 3) == ta.lowest(low, 8) and avgLowWick > 2.5*avgHighWick
alertcondition(bullish3WicksSig, title='DRE Bullish 3 wicks sig', message='bullish wicks')
plotshape(bullish3WicksSig and iBullsh3Wicks, title='bullish wicks', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='wicks', textcolor=color.new(color.green, 0))



alertcondition(topSig or bearishKickerSig or bearishEngulfingSig or bearish3WicksSig, title='DRE Bearish pattern sig', message='bearish pattern')
alertcondition(bullishKickerSig or bullishEngulfingSig or bullish3WicksSig, title='DRE Bullish pattern sig', message='bullish pattern')
