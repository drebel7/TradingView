func_sig_parabolic_top(volumeMaLen, closeMaLen, volumeDownDaysAfterTop, closeDownDaysAfterTop, prevPeriodGainLen, prevPeriodGainMultiple) =>
    volMa = ta.sma(volume, volumeMaLen)
    closeMa = ta.sma(close, closeMaLen)
    barsSinceVolDown = ta.barssince(volMa < volMa[3])
    ret = barsSinceVolDown > 15 and close > prevPeriodGainMultiple * ta.lowest(close, prevPeriodGainLen) and volume < volume[volumeDownDaysAfterTop] and open < close[closeDownDaysAfterTop]
    ret

func_crossover(_source) =>
    math.min(low, close[1]) <= _source and close > _source
func_crossunder(_source) =>
    math.max(high, close[1]) >= _source and close < _source

iJunior = input.bool(defval = true, title = 'bullish', group = 'junior')
iJuniorMaxAge = input.int(defval = 325, title = 'Max junior age', minval = 20, maxval = 500, group = 'junior')

func_init_vwap_excl_init_higher_volume() =>
    x = 0.5
    reset = bar_index == 0 or bar_index == 1 and volume < x * volume[1] or bar_index == 2 and volume < x * volume[1] and volume[1] < x * volume[2]
    ret = ta.vwap(ohlc4, reset)
    ret

func_sig_junior(_maxAge) =>
    vwap = func_init_vwap_excl_init_higher_volume()
    ret = bar_index < _maxAge and func_crossover(vwap)
    ret

iCoil = input.bool(defval = true, title = 'bullish', group = 'coil_n_strech', inline = 'coil1')

func_ma_distance(_shortMaLen, _mediumMaLen, _longMaLen) =>
    ma1 = ta.sma(close, _shortMaLen)
    ma2 = ta.sma(close, _mediumMaLen)
    ma3 = ta.sma(close, _longMaLen)
    vwap = ta.vwap(ohlc4, bar_index == 0)
    usedLongMaLen = bar_index < _longMaLen ? bar_index : _longMaLen
    ma3used = bar_index < _longMaLen ? vwap : ma3
    lenSum = _shortMaLen + _mediumMaLen + usedLongMaLen
    distance = math.sqrt((lenSum - _shortMaLen) * math.pow(ma1 - close, 2) + (lenSum - _mediumMaLen) * math.pow(ma2 - close, 2) + (lenSum - usedLongMaLen) * math.pow(ma3used - close, 2))
    distance

func_sig_coil(_dist, _longMaLen, _mediumMaLen, _distPercentRank) =>
    percentRankLen = bar_index < 1 ? 1 : math.min(bar_index, 3 * _longMaLen)
    ret = ta.percentrank(_dist, percentRankLen) < _distPercentRank and func_crossover(ta.sma(close, _mediumMaLen))
    ret

func_sig_strech(_dist, _longMaLen, _distPercentRank) =>
    percentRankLen = bar_index < 1 ? 1 : math.min(bar_index, 3 * _longMaLen)
    ret = ta.percentrank(_dist, percentRankLen) > 100 - _distPercentRank and close > close[_longMaLen] and close > close[1]
    ret

func_sig_starter(_isBullish, _maLen, _minMove1Len, _maxMove2Len, _smoothing) =>
    ma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    move1 = math.min(1000, ta.barssince((_isBullish ? ma > ma[1] : ma < ma[1]) or bar_index < _maLen))
    move2 = math.min(1000, ta.barssince(move1 == 0 and (_isBullish ? move1[1] > _minMove1Len : move1[1] < _minMove1Len)))
    ret = move2 < _maxMove2Len and (_isBullish ? func_crossover(ma) and ma >= ma[move2] : func_crossunder(ma) and ma <= ma[move2])
    ret

func_sig_f_starter(_maLen, _minMove1Len, _atrMulti, _smoothing) =>
    sma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    vwma = ta.vwma(ohlc4, _maLen)
    move1 = ta.barssince(sma > sma[1])
    maDiffLen = math.ceil(0.4 * (move1 > _minMove1Len ? move1 : _minMove1Len))
    highestMaDiff = ta.highest(sma - vwma, maDiffLen) / mainAtr
    ret = move1 > _minMove1Len and (func_crossover(sma) or func_crossover(vwma)) and highestMaDiff > _atrMulti and highestMaDiff > highestMaDiff[maDiffLen]
    ret

    ret = _shortMa > _shortMa[1] and _longMa > _longMa[1] and _shortMa > _longMa and func_crossover(_shortMa + _offset * _atr)
    ret

func_sig_bull_sl_hunter(_atr, _lookback, _gracePeriod) =>
    localLow = ta.lowest(low, _lookback)
    localLowCandlesAgo = ta.barssince(low == localLow)
    localHigh = ta.highest(high, math.max(1, nz(localLowCandlesAgo)))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = func_crossover(localLow[_gracePeriod]) and localHigh[_gracePeriod] > localLow[_gracePeriod] + lohi * _atr and localLow[_gracePeriod] == localLow[2 * _gracePeriod]
    sig

func_sig_bear_sl_hunter(_atr, _lookback, _gracePeriod) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localLow = ta.lowest(low, math.max(1, nz(localHighCandlesAgo)))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = func_crossunder(localHigh[_gracePeriod]) and localLow[_gracePeriod] < localHigh[_gracePeriod] - lohi * _atr and localHigh[_gracePeriod] == localHigh[2 * _gracePeriod]
    sig

iBullishStepper = input.bool(defval = true, title = 'bullish', group = 'stepper', inline = 'stepper1')

func_sig_bull_stepper(_atr, _mainMa, _lookback, _maxMaDist) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localHighVwap = ta.vwap(ohlc4, high == localHigh)
    maDist = localHighVwap - _mainMa
    sig = _mainMa > _mainMa[1] and localHigh[localHighCandlesAgo] > _mainMa[localHighCandlesAgo] + 1.5 * math.log(_lookback) * _atr and maDist > 0 and maDist < _maxMaDist * _atr and func_crossover(localHighVwap)
    sig

func_sig_bull_squeezer(_atr, _lookback, _gracePeriod, _minInitVwapDist, _maxVwapDist) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localLow = ta.lowest(low, math.max(1, nz(localHighCandlesAgo)))
    localLowCandlesAgo = ta.barssince(low == localLow)
    localHighVwap = ta.vwap(ohlc4, high == localHigh)
    localLowVwap = ta.vwap(ohlc4, low == localLow)
    maDist = localHighVwap - localLowVwap
    sig = localLowCandlesAgo > 50 and localHighVwap[localLowCandlesAgo] > localLow[localLowCandlesAgo] + _minInitVwapDist * _atr and maDist > 0 and maDist < _maxVwapDist * _atr and func_crossover(localHighVwap) and localHigh == localHigh[_gracePeriod]
    sig
