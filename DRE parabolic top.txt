func_init_vwap_excl_init_higher_volume() =>
    x = 0.5
    reset = bar_index == 0 or bar_index == 1 and volume < x * volume[1] or bar_index == 2 and volume < x * volume[1] and volume[1] < x * volume[2]
    ret = ta.vwap(ohlc4, reset)
    ret

func_sig_junior(_maxAge) =>
    vwap = func_init_vwap_excl_init_higher_volume()
    ret = bar_index < _maxAge and drelib.crossover(vwap)
    ret

func_ma_distance(_shortMaLen, _mediumMaLen, _longMaLen) =>
    ma1 = ta.sma(close, _shortMaLen)
    ma2 = ta.sma(close, _mediumMaLen)
    ma3 = ta.sma(close, _longMaLen)
    vwap = ta.vwap(ohlc4, bar_index == 0)
    usedLongMaLen = bar_index < _longMaLen ? bar_index : _longMaLen
    ma3used = bar_index < _longMaLen ? vwap : ma3
    lenSum = _shortMaLen + _mediumMaLen + usedLongMaLen
    distance = math.sqrt((lenSum - _shortMaLen) * math.pow(ma1 - close, 2) + (lenSum - _mediumMaLen) * math.pow(ma2 - close, 2) + (lenSum - usedLongMaLen) * math.pow(ma3used - close, 2))
    distance

func_sig_coil(_dist, _longMaLen, _mediumMaLen, _distPercentRank) =>
    percentRankLen = bar_index < 1 ? 1 : math.min(bar_index, 3 * _longMaLen)
    mediumMa = ta.sma(close, _mediumMaLen)
    ret = ta.percentrank(_dist, percentRankLen) < _distPercentRank and drelib.crossover(mediumMa)
    ret

func_sig_strech(_dist, _longMaLen, _distPercentRank) =>
    percentRankLen = bar_index < 1 ? 1 : math.min(bar_index, 3 * _longMaLen)
    ret = ta.percentrank(_dist, percentRankLen) > 100 - _distPercentRank and close > close[_longMaLen] and close > close[1]
    ret

func_sig_starter(_isBullish, _maLen, _minMove1Len, _maxMove2Len, _smoothing) =>
    ma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    // Only calculate move1 after MA has enough data
    move1 = bar_index >= _maLen ? math.min(1000, ta.barssince(_isBullish ? ma > ma[1] : ma < ma[1])) : 1000
    move2 = math.min(1000, ta.barssince(move1 == 0 and (_isBullish ? move1[1] > _minMove1Len : move1[1] < _minMove1Len)))
    ret = move2 < _maxMove2Len and (_isBullish ? drelib.crossover(ma) and ma >= ma[move2] : drelib.crossunder(ma) and ma <= ma[move2])
    ret

func_sig_f_starter(_maLen, _minMove1Len, _atrMulti, _smoothing) =>
    sma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    vwma = ta.vwma(ohlc4, _maLen)
    move1 = ta.barssince(sma > sma[1])
    maDiffLen = math.ceil(0.4 * (move1 > _minMove1Len ? move1 : _minMove1Len))
    highestMaDiff = ta.highest(sma - vwma, maDiffLen) / mainAtr
    ret = move1 > _minMove1Len and (drelib.crossover(sma) or drelib.crossover(vwma)) and highestMaDiff > _atrMulti and highestMaDiff > highestMaDiff[maDiffLen]
    ret

func_sig_bull_sl_hunter(_atr, _lookback, _gracePeriod) =>
    localLow = ta.lowest(low, _lookback)
    localLowCandlesAgo = ta.barssince(low == localLow)
    validCandlesAgo = nz(localLowCandlesAgo, _lookback)
    localHigh = ta.highest(high, math.max(1, validCandlesAgo))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = drelib.crossover(localLow[_gracePeriod]) and localHigh[_gracePeriod] > localLow[_gracePeriod] + lohi * _atr and localLow[_gracePeriod] == localLow[2 * _gracePeriod]
    sig

func_sig_bear_sl_hunter(_atr, _lookback, _gracePeriod) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    validCandlesAgo = nz(localHighCandlesAgo, _lookback)
    localLow = ta.lowest(low, math.max(1, validCandlesAgo))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = drelib.crossunder(localHigh[_gracePeriod]) and localLow[_gracePeriod] < localHigh[_gracePeriod] - lohi * _atr and localHigh[_gracePeriod] == localHigh[2 * _gracePeriod]
    sig

func_sig_bull_stepper(_atr, _mainMa, _lookback, _maxMaDist) =>
    localHigh = ta.highest(high, _lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)

    sig

func_rmv(_artShortLen1, _artShortLen2, _artShortLen3, _MainLookback) =>
    shortAtrAvg = (ta.atr(_artShortLen1) + ta.atr(_artShortLen2) + ta.atr(_artShortLen3)) / 3
    highestShortAvg = ta.highest(shortAtrAvg, _MainLookback)
    lowestShortAvg = ta.lowest(shortAtrAvg, _MainLookback)
    result = (shortAtrAvg - lowestShortAvg) / math.max(highestShortAvg - lowestShortAvg, 0.001) * 100
    result
