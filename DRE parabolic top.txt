import drebel/DRELib/1 as drelib

iParabolicTop = input.bool(defval=true, title='Parabolic top', group='parabolic top')
iMultipleGain = input.float(defval=1.6, title='Multiple Gain Since Low', minval=1.2, maxval=4, step=0.2, group='parabolic top')
iMultipleInDays = input.int(defval=100, title='Number Of Days To Multiple', minval=5, maxval=150, step=5, group='parabolic top')
iDownDaysAfterTop = input.int(defval=2, title='Down Days After Top', minval=1, maxval=5, group='parabolic top')
iVolDownDaysAfterTop = input.int(defval=2, title='Down Days After Top (Volume)', minval=1, maxval=5, group='parabolic top')

func_sig_parabolic_top(volumeMaLen, closeMaLen, volumeDownDaysAfterTop, closeDownDaysAfterTop, prevPeriodGainLen, prevPeriodGainMultiple) =>
    volMa = ta.sma(volume, volumeMaLen)
    closeMa = ta.sma(close, closeMaLen)
    barsSinceVolDown = ta.barssince(volMa < volMa[3])
    ret = barsSinceVolDown > 15 and close > prevPeriodGainMultiple * ta.lowest(close, prevPeriodGainLen) and volume < volume[volumeDownDaysAfterTop] and open < close[closeDownDaysAfterTop]

topSig = func_sig_parabolic_top(30, 20, iVolDownDaysAfterTop, iDownDaysAfterTop, iMultipleInDays, iMultipleGain)

alertcondition(topSig, title='DRE Parabolic TOP sig', message='parabolic TOP')
plotshape(topSig and iParabolicTop, title='parabolic top', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='top', textcolor=color.new(color.red, 0))


func_crossover(middle) =>
    math.min(low, close[1]) <= middle and close > middle
func_crossunder(middle) =>
    math.max(high, close[1]) >= middle and close < middle

iJunior = input.bool(defval=true, title='bullish', group='junior')
iJuniorMaxAge = input.int(defval=250, title='Max junior age', minval=20, maxval=500, group='junior')

func_sig_junior(maxAge) =>
    x = 0.5
    reset = bar_index == 0 or (bar_index == 1 and volume < x * volume[1]) or (bar_index == 2 and volume < x * volume[1] and volume[1] < x * volume[2])
    vwap = ta.vwap(ohlc4, reset)
    ret = bar_index < maxAge and func_crossover(vwap)

juniorSig = func_sig_junior(iJuniorMaxAge)

alertcondition(juniorSig, title='DRE junior sig', message='junior')
plotshape(juniorSig and iJunior, title='junior', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='junior', textcolor=color.new(color.green, 0))


iCoil = input.bool(defval=true, title='bullish', group='coil_n_strech', inline='coil1')
iCoilStrech = input.bool(defval=false, title='bearish', group='coil_n_strech', inline='coil1')
iCoilShortMaLength = input.int(defval=50, title='Short MA length', minval=10, maxval=200, group='coil_n_strech')
iCoilMediumMaLength = input.int(defval=200, title='Short MA length', minval=15, maxval=500, group='coil_n_strech')
iCoilLongMaLength = input.int(defval=750, title='Long MA length', minval=20, maxval=1000, group='coil_n_strech')
iCoilPercent = input.int(defval=2, title='Percentile rank distance from MAs', minval=1, maxval=20, group='coil_n_strech')

func_ma_distance(shortMaLen, mediumMaLen, longMaLen) =>
    shortMa = ta.sma(close, shortMaLen)
    mediumMa = ta.sma(close, mediumMaLen)
    longMa = ta.sma(close, longMaLen)
    lenSum = shortMaLen + mediumMaLen + longMaLen
    distance = math.sqrt((lenSum-shortMaLen)*math.pow(shortMa - close, 2) + (lenSum-mediumMaLen)*math.pow(mediumMa - close, 2) + (lenSum-longMaLen)*math.pow(longMa - close, 2))

func_sig_coil(distance, longMaLen, mediumMaLen, distancePercentRank) =>
    ret = ta.percentrank(distance, 3 * longMaLen) < distancePercentRank and func_crossover(ta.sma(close, mediumMaLen))

func_sig_strech(distance, longMaLen, distancePercentRank) =>
    ret = ta.percentrank(distance, 3 * longMaLen) > 100 - distancePercentRank and close > close[longMaLen] and close > close[1]

coilBullishSig = func_sig_coil(func_ma_distance(iCoilShortMaLength, iCoilMediumMaLength, iCoilLongMaLength), iCoilLongMaLength, iCoilMediumMaLength, iCoilPercent)
strechBearishSig = func_sig_strech(func_ma_distance(iCoilShortMaLength, iCoilMediumMaLength, iCoilLongMaLength), iCoilLongMaLength, iCoilPercent)

alertcondition(coilBullishSig, title='DRE coil sig', message='coil')
alertcondition(strechBearishSig, title='DRE strech sig', message='strech')
plotshape(coilBullishSig and iCoil, title='coil', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='coil', textcolor=color.new(color.green, 0))
plotshape(strechBearishSig and iCoilStrech, title='strech', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='strech', textcolor=color.new(color.red, 0))


iBullishStarter = input.bool(defval=true, title='bullish', group='starter', inline = 'starter1')
iBearishStarter = input.bool(defval=false, title='bearish', group='starter', inline = 'starter1')
iBullishFireStarter = input.bool(defval=true, title='bullish fire starter', group='starter')
iStarterMinDecliningDays = input.int(defval=200, title='Number of SMA declining days', minval=20, maxval=1000, group='starter')
iStarterMaxIncliningDays = input.int(defval=200, title='Number of SMA inclining day', minval=0, maxval=300, group='starter')
iStarterSmaSmoothing = input.int(defval = 3, title='SMA smoothing', minval=0, maxval=10, group='starter')
iFireStarterMaDiffAtrMultiple = input.float(defval = 1, title='Fire starter MA difference ATR multiple', minval=0.5, maxval=10, step=0.5, group='starter')

func_sig_starter(isBullish, maLen, minMove1Len, maxMove2Len, isSmoothing) =>
    ma = isSmoothing > 0 ? ta.ema(ta.sma(close, maLen), isSmoothing) : ta.sma(close, maLen)
    move1 = math.min(1000, ta.barssince((isBullish ? ma > ma[1] : ma < ma[1]) or bar_index < maLen))
    move2 = math.min(1000, ta.barssince(move1 == 0 and (isBullish ? move1[1] > minMove1Len : move1[1] < minMove1Len)))
    ret = move2 < maxMove2Len and (isBullish ? func_crossover(ma) and ma >= ma[move2] : func_crossunder(ma) and ma <= ma[move2])

func_sig_f_starter(maLen, minMove1Len, atrMultiple, isSmoothing) =>
    sma = isSmoothing > 0 ? ta.ema(ta.sma(close, maLen), isSmoothing) : ta.sma(close, maLen)
    vwma = ta.vwma(ohlc4, maLen)
    atr = ta.atr(maLen)
    move1 = ta.barssince(sma > sma[1])
    maDiffLen = math.ceil(0.4 * (move1 > minMove1Len ? move1 : minMove1Len))
    highestMaDiff = ta.highest(sma - vwma, maDiffLen) / atr
    ret = move1 > minMove1Len and (func_crossover(sma) or func_crossover(vwma)) and highestMaDiff > atrMultiple and highestMaDiff > highestMaDiff[maDiffLen]

bullishStarterSig = func_sig_starter(true, 200, iStarterMinDecliningDays, iStarterMaxIncliningDays, iStarterSmaSmoothing)
bearishStarterSig = func_sig_starter(false, 200, iStarterMinDecliningDays, iStarterMaxIncliningDays, iStarterSmaSmoothing)
bullishFireStarterSig = func_sig_f_starter(200, iStarterMinDecliningDays, iFireStarterMaDiffAtrMultiple, iStarterSmaSmoothing)

alertcondition(bullishStarterSig, title='DRE bullish starter sig', message='bullish starter')
alertcondition(bearishStarterSig, title='DRE bearish starter sig', message='bearish starter')
alertcondition(bullishFireStarterSig, title='DRE bullish fire starter sig', message='bullish fire starter')
plotshape(bullishStarterSig and iBullishStarter, title='bullish starter', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='starter', textcolor=color.new(color.green, 0))
plotshape(bearishStarterSig and iBearishStarter, title='bearish starter', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='starter', textcolor=color.new(color.red, 0))
plotshape(bullishFireStarterSig and iBullishFireStarter, title='bullish fire starter', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='f.starter', textcolor=color.new(color.green, 0))


iLonger = input.bool(defval=true, title='bullish', group='longer', inline = 'longer1')
iLongerShortMaLength = input.int(defval=200, title='Short MA length', minval=10, maxval=500, group='longer')
iLongerLongMaLength = input.int(defval=750, title='Long MA length', minval=20, maxval=1000, group='longer')
iLongerOffset = input.float(defval=0.1, title='Price offset (atr multiple)', minval=-2.0, maxval=2.0, step=0.5, group='longer')

func_sig_longer(shortMaLength, longMaLength, offset) =>
    shortMa = ta.sma(close, shortMaLength)
    longMa = ta.sma(close, longMaLength)
    atr = ta.atr(shortMaLength)
    ret = shortMa > shortMa[1] and longMa > longMa[1] and shortMa > longMa and func_crossover(shortMa + offset * atr)

longerSig = func_sig_longer(iLongerShortMaLength, iLongerLongMaLength, iLongerOffset)

alertcondition(longerSig, title='DRE longer sig', message='longer')
plotshape(longerSig and iLonger, title='longer', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='longer', textcolor=color.new(color.green, 0))


iBullishZscore = input.bool(defval=false, title='bullish', group='zscore', inline = 'zscore1')
iBearishZscore = input.bool(defval=false, title='bearish', group='zscore', inline = 'zscore1')
iZScoreMaLenth = input.int(200, "Z-Score MA Length (long)", minval=20, maxval=1000, group = "zscore")
iZScore2ndMaLength = input.int(21, "Z-Score 2nd MA Length (short)", minval=3, maxval=21, group = "zscore")
iZScore2ndMaMinScore = input.float(2.3, "Min. zScore", minval=1.0, maxval=14, step=0.1, group = "zscore")

zscoreVolCond = ta.ema(drelib.z_score(volume, iZScoreMaLenth), iZScore2ndMaLength) >= iZScore2ndMaMinScore
zscoreEma = ta.ema(close, 2 * iZScore2ndMaLength / 3)
zscoreBullishSig = zscoreVolCond and zscoreEma > zscoreEma[1]
zscoreBearishSig = zscoreVolCond and zscoreEma < zscoreEma[1]

alertcondition(zscoreBullishSig, title='DRE bubble up', message='bubble up')
alertcondition(zscoreBearishSig, title='DRE bubble dn', message='bubble dn')
plotshape(zscoreBullishSig and iBullishZscore, title='bubble up', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='bubble up', textcolor=color.new(color.black, 0))
plotshape(zscoreBearishSig and iBearishZscore, title='bubble dn', location=location.abovebar, color=color.new(color.red, 0), style=shape.triangledown, text='bubble dn', textcolor=color.new(color.black, 0))


iBullishHunter = input.bool(defval=false, title='bullish', group='SL hunter', inline = 'hunter1')
//iBearishHunter = input.bool(defval=false, title='bearish', group='SL hunter', inline = 'hunter1')
iHunterMinTimeDistance = input.int(defval=50, title='Min. time distance', minval=20, maxval=750, group='SL hunter')
iHunterMaxTimeDistance = input.int(defval=500, title='Max. time distance', minval=50, maxval=1000, group='SL hunter')
iHunterMinBounceSteepness = input.float(defval=0.05, title='Min steepness from 1st bounce', minval=0.01, maxval=1.0, step=0.01, group='SL hunter')

atr200 = ta.atr(200)
localLow = ta.lowest(low, iHunterMaxTimeDistance)
localLowCandlesAgo = ta.barssince(low == localLow)
localHigh = ta.highest(high, math.max(1, nz(localLowCandlesAgo)))
localHighCandlesAgo = ta.barssince(high == localHigh)
steepness = (localHighCandlesAgo[iHunterMinTimeDistance] - localLowCandlesAgo[iHunterMinTimeDistance]) / atr200[iHunterMinTimeDistance]

bullishHunterSig = func_crossover(localLow[iHunterMinTimeDistance]) and localHigh[iHunterMinTimeDistance] > localLow[iHunterMinTimeDistance] and steepness > iHunterMinBounceSteepness

alertcondition(bullishHunterSig, title='DRE Bullish SL hunter', message='sl.hunter')
plotshape(bullishHunterSig and iBullishHunter, title='bullish sl hunter', location=location.belowbar, color=color.new(color.green, 0), style=shape.triangleup, text='sl.hunter', textcolor=color.new(color.green, 0))


alertcondition(juniorSig or bullishStarterSig or coilBullishSig or longerSig or zscoreBullishSig, title='DRE Bullish pattern sig', message='bullish pattern')
alertcondition(topSig or bearishStarterSig or strechBearishSig or zscoreBearishSig, title='DRE Bearish pattern sig', message='bearish pattern')

alertcondition(juniorSig or bullishStarterSig or coilBullishSig or longerSig or topSig or bearishStarterSig or strechBearishSig or zscoreBullishSig or zscoreBearishSig, title='DRE ALL pattern sig', message='all patterns')
