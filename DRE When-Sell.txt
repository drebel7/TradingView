//@version=6
indicator(title = 'DRE When-Sell', overlay = true)

import drebel/DRELib/4 as drelib

iSmaLen1 = input.int(defval = 10, title = 'SMA length 1', minval = 1, group = 'Moving averages')
iSmaLen2 = input.int(defval = 20, title = 'SMA length 2', minval = 1, group = 'Moving averages')
iSmaLen3 = input.int(defval = 50, title = 'SMA length 3', minval = 1, group = 'Moving averages')

iShowParabolicRiseStop = input.bool(defval = true, title = 'Show break of LOD', group = 'Parabolic rise')
iParabolicRisePercentile = input.int(defval = 99, title = 'Percentile above sma 1/2/3', minval = 0, maxval = 100, group = 'Parabolic rise')
iParabolicRiseBigRedCandle = input.int(defval = 230, title = 'Min. ADR[%] LOD break', minval = 0, maxval = 500, step = 10, group = 'Parabolic rise')
iParabolicRiseVolumeCond = input.bool(defval = true, title = 'Highest volume of last 10 days within last 3 days', group = 'Parabolic rise')
iShowParabolicTop = input.bool(defval = true, title = 'Show parabolic top')
iShowCloseBelowSma1 = input.bool(defval = false, title = 'Show close below sma1')
iShowCloseBelowSma2 = input.bool(defval = false, title = 'Show close below sma2')
iShowCloseBelowSma3 = input.bool(defval = false, title = 'Show close below sma3')

sma1 = ta.sma(close, iSmaLen1)
sma2 = ta.sma(close, iSmaLen2)
sma3 = ta.sma(close, iSmaLen3)
adr50 = ta.sma(high - low, 50)
volHighest10 = ta.highest(volume, 10)
volMa = ta.sma(volume, 30)
atr200 = ta.atr(200)

//candle parameters
highShadow = high != low ? (high - math.max(open, close)) / (high-low) : 0
candleBody = high != low ? (math.max(open, close) - math.min(open, close)) / (high-low) : 0
lowShadow = high != low ? (math.min(open, close) - low) / (high-low) : 0
candleColor = close > open ? 1 : close < open ? -1 : 0

isMarubozuGreen = candleBody > 0.75 and candleColor == 1

//parabolic rise LOD break
aboveSma1PctCond = ta.percentrank(high - sma1, 500) >= iParabolicRisePercentile
aboveSma2PctCond = ta.percentrank(high - sma2, 500) >= iParabolicRisePercentile
aboveSma3PctCond = ta.percentrank(high - sma3, 500) >= iParabolicRisePercentile
volumeCond = iParabolicRiseVolumeCond and (volume == volHighest10 or volume[1] == volHighest10[1] or volume[2] == volHighest10[2]) or not iParabolicRiseVolumeCond

parabolicRiseCond = aboveSma1PctCond and aboveSma2PctCond and aboveSma3PctCond and close > 1.7 * close[100]
parabolicRiseLodBreak = (parabolicRiseCond[1] or parabolicRiseCond) and low < low[1] and (high - low) > adr50 * (iParabolicRiseBigRedCandle / 100) and not isMarubozuGreen
parabolicRiseLodBreakCond = parabolicRiseLodBreak and not parabolicRiseLodBreak[1] and not parabolicRiseLodBreak[2] and not parabolicRiseLodBreak[3] and volumeCond

//parabolic top
top1Sig = drelib.sig_parabolic_top(volMa, 2, 100, 8, atr200)
top2Sig = drelib.sig_parabolic_top(volMa, 3, 100, 8, atr200)
top3Sig = drelib.sig_parabolic_top(volMa, 4, 100, 8, atr200)
topSig = top1Sig or top2Sig or top3Sig


alertcondition(parabolicRiseLodBreakCond, title = 'parabolic rise lod - break', message = 'LOD break')
if iShowParabolicRiseStop and parabolicRiseLodBreakCond
    label.new(bar_index, low[1], tooltip = 'parabolic rise lod - break', color = color.new(color.red, 70), style = label.style_circle, size = size.tiny)
//plotshape(iShowParabolicRiseStop and parabolicRiseLodBreakCond, title = 'parabolic rise lod - break', location = location.absolute, color = color.new(color.red, 0), style = shape.triangledown, text = 'par. rise LOD break', textcolor = color.new(color.red, 0))

alertcondition(topSig, title = 'parabolic top', message = 'top')
if iShowParabolicTop and topSig
    label.new(bar_index, close, tooltip = 'top', color = color.new(color.red, 70), style = label.style_circle, size = size.tiny)
//plotshape(iShowParabolicTop and topSig ? close : na, title = 'parabolic top', location = location.absolute, color = color.new(color.red, 0), style = shape.triangledown, text = 'top', textcolor = color.new(color.red, 0))

belowSma1 = close < sma1
newBelowSma1 = belowSma1 and not belowSma1[1]
alertcondition(newBelowSma1, title = 'close below sma1', message = 'close < sma1')
if iShowCloseBelowSma1 and newBelowSma1
    label.new(bar_index, close, tooltip = 'close < sma1', color = color.new(color.red, 70), style = label.style_circle, size = size.tiny)
//plotshape(iShowCloseBelowSma1 and newBelowSma1, title = 'close below sma1', location = location.absolute, color = color.new(color.red, 0), style = shape.triangledown, text = 'close < sma1', textcolor = color.new(color.red, 0))

belowSma2 = close < sma2
newBelowSma2 = belowSma2 and not belowSma2[1]
alertcondition(newBelowSma2, title = 'close below sma2', message = 'close < sma2')
if iShowCloseBelowSma2 and newBelowSma2
    label.new(bar_index, close, tooltip = 'close < sma2', color = color.new(color.red, 70), style = label.style_circle, size = size.tiny)
//plotshape(iShowCloseBelowSma2 and newBelowSma2, title = 'close below sma2', location = location.absolute, color = color.new(color.red, 0), style = shape.triangledown, text = 'close < sma2', textcolor = color.new(color.red, 0))

belowSma3 = close < sma3
newBelowSma3 = belowSma3 and not belowSma3[1]
alertcondition(newBelowSma3, title = 'close below sma3', message = 'close < sma3')
if iShowCloseBelowSma3 and newBelowSma3
    label.new(bar_index, close, tooltip = 'close < sma3', color = color.new(color.red, 70), style = label.style_circle, size = size.tiny)
//plotshape(iShowCloseBelowSma3 and newBelowSma3, title = 'close below sma3', location = location.absolute, color = color.new(color.red, 0), style = shape.triangledown, text = 'close < sma3', textcolor = color.new(color.red, 0))

dttTimePct = 100 * (math.min(timenow, time_close) - time) / (time_close - time)
onSessionCloseConditions = dttTimePct == 100 and (topSig or newBelowSma1 or newBelowSma2 or newBelowSma3)
allCond = parabolicRiseLodBreakCond or onSessionCloseConditions
alertcondition(allCond, title = 'when-sell - all sigs', message = 'when-sell')
