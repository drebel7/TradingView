export getDaysOnExchange() =>
    maxIndex = math.min(bar_index, 500)
    (timenow - time[maxIndex]) * bar_index / maxIndex / 1000 / 60 / 60 / 24

export getWeeksOnExchange() =>
    getDaysOnExchange() / 7

export getMonthsOnExchange() =>
    getDaysOnExchange() / 30.43

export getYearsOnExchange() =>
    getDaysOnExchange() / 365.25

// from MLExtensions by jdehorty
// @function Rescales a source value with an unbounded range to a target range.
// @param src <series float> The input series
// @param min <float> The minimum value of the unbounded range
// @param max <float> The maximum value of the unbounded range
// @returns <series float> The normalized series
export normalize(series float src, float min, float max) =>
    var _historicMin = 10e10
    var _historicMax = -10e10
    _historicMin := math.min(nz(src, _historicMin), _historicMin)
    _historicMax := math.max(nz(src, _historicMax), _historicMax)
    min + (max - min) * (src - _historicMin) / math.max(_historicMax - _historicMin, 10e-10)

// from MLExtensions by jdehorty
// @function Rescales a source value with a bounded range to anther bounded range
// @param src <series float> The input series
// @param oldMin <float> The minimum value of the range to rescale from
// @param oldMax <float> The maximum value of the range to rescale from
// @param newMin <float> The minimum value of the range to rescale to
// @param newMax <float> The maximum value of the range to rescale to 
// @returns <series float> The rescaled series
export rescale(series float src, float oldMin, float oldMax, float newMin, float newMax) =>
    newMin + (newMax - newMin) * (src - oldMin) / math.max(oldMax - oldMin, 10e-10)


export z_score(series float _src, simple int _length) =>
    _value = (_src - ta.ema(_src, _length)) / ta.stdev(_src - ta.ema(_src, _length), _length)
    _value

export n_z_score(series float src, simple int n1, simple int n2) =>
    normalize(ta.ema(z_score(src, n1), n2), 0, 1)

// @function indicates when a candle crosses given source upwards
// @param _src <series float> The input series 
// @returns <series bool> true when source is crossed by a candle
export candle_crossover(series float _src) =>
    math.min(low, close[1]) <= _src and close > _src

// @function indicates when a candle crosses given source downwards
// @param _src <series float> The input series 
// @returns <series bool> true when source is crossed by a candle
export candle_crossunder(series float _src) =>
    math.max(high, close[1]) >= _src and close < _src

export ma_distance(int _maLen) =>
    ta.sma(close, _maLen) - close

export ma_distance(int _shortMaLen, int _longMaLen) =>
    shortMa = ta.sma(close, _shortMaLen)
    longMa = ta.sma(close, _longMaLen)
    lenSum = _shortMaLen + _longMaLen
    math.sqrt((lenSum - _shortMaLen) * math.pow(shortMa - close, 2) + (lenSum - _longMaLen) * math.pow(longMa - close, 2))

export ma_distance(int _shortMaLen, int _mediumMaLen, int _longMaLen) =>
    shortMa = ta.sma(close, _shortMaLen)
    mediumMa = ta.sma(close, _mediumMaLen)
    longMa = ta.sma(close, _longMaLen)
    lenSum = _shortMaLen + _mediumMaLen + _longMaLen
    math.sqrt((lenSum - _shortMaLen) * math.pow(shortMa - close, 2) + (lenSum - _mediumMaLen) * math.pow(mediumMa - close, 2) + (lenSum - _longMaLen) * math.pow(longMa - close, 2))

export price_pct(series float _src) =>
    100 * (_src - close) / close

export crossover(series float _source) =>
    math.min(low, close[1]) <= _source and close > _source

export crossover2(series float _source1, series float _source2) =>
    _source1[1] <= _source2[1] and _source1 > _source2

export crossunder(series float _source) =>
    math.max(high, close[1]) >= _source and close < _source

export crossunder2(series float _source1, series float _source2) =>
    _source1[1] >= _source2[1] and _source1 < _source2

export priceMasTrendAligned(series float closePrice, series float lowPrice, series float highPrice, int _tradingDaysInAYear) =>
    ma200 = ta.sma(closePrice, 200)
    ma150 = ta.sma(closePrice, 150)
    ma50 = ta.sma(closePrice, 50)
    yLow = ta.lowest(lowPrice, _tradingDaysInAYear)
    yHigh = ta.highest(highPrice, _tradingDaysInAYear)
    ma200GoingUp_per35days = math.sum(ma200 > ma200[1] ? 1 : 0, 35)
    ma200GoingUpCond = ma200GoingUp_per35days > 30
    masAlignedBullishCond = ma50 > ma150 and ma50 > ma200 and ma150 > ma200 and ma200GoingUpCond
    priceAlignedBullishCond = closePrice > 1.25 * yLow and closePrice > yHigh / 1.25
    priceMasAlignedBullishCond = closePrice > ma50 / 1.08 and closePrice > ma150 and closePrice > ma200
    bulishTrend = masAlignedBullishCond and priceAlignedBullishCond and priceMasAlignedBullishCond
    ma200GoingDnCond = ma200GoingUp_per35days < 5
    masAlignedBearishCond = ma50 < ma150 and ma50 < ma200 and ma150 < ma200 and ma200GoingDnCond
    priceAlignedBearishCond = closePrice < 1.25 * yLow and closePrice < yHigh / 1.25
    priceMasAlignedBearishCond = closePrice < ma50 * 1.08 and closePrice < ma150 and closePrice < ma200
    bearishTrend = masAlignedBearishCond and priceAlignedBearishCond and priceMasAlignedBearishCond
    [bulishTrend, bearishTrend]

export simpleRelativeStrengthStrong(int _tradingDaysInAYear) =>
    benchmark = getDefaultBenchmark(close)
    halfYearDays = _tradingDaysInAYear / 2
    quarterYearDays = _tradingDaysInAYear / 4
    relativelyStrong = close / benchmark > close[halfYearDays] / benchmark[halfYearDays] and close / benchmark > close[quarterYearDays] / benchmark[quarterYearDays]
    relativelyStrong

export simpleRelativeStrengthWeak(int _tradingDaysInAYear) =>
    benchmark = getDefaultBenchmark(close)
    halfYearDays = _tradingDaysInAYear / 2
    quarterYearDays = _tradingDaysInAYear / 4
    relativelyWeak = close / benchmark < close[halfYearDays] / benchmark[halfYearDays] and close / benchmark < close[quarterYearDays] / benchmark[quarterYearDays]
    relativelyWeak

earningsGoingUp(_eps, int _period) =>
    earningsDay = _eps != _eps[1]
    earningsDayCountPeriod = math.sum(earningsDay ? 1 : 0, _period)
    epsBetter_10 = earningsDay and _eps > _eps[getDefaultStockTradingDaysInAYear()] ? 1 : 0
    epsBetterInPeriod = math.sum(epsBetter_10, _period)
    [epsBetterInPeriod, earningsDayCountPeriod]

earningsGoingUp(_epsAct, _epsStd, int _period) =>
    [actEpsBetter1Y, actEps1Y] = earningsGoingUp(_epsAct, _period)
    [stdEpsBetter1Y, stdEps1Y] = earningsGoingUp(_epsStd, _period)
    earningsDayCountPeriod = nz(actEps1Y) > nz(stdEps1Y) ? nz(actEps1Y) : nz(stdEps1Y)
    epsBetterInPeriod = nz(actEps1Y) > nz(stdEps1Y) ? nz(actEpsBetter1Y) : nz(stdEpsBetter1Y)
    [epsBetterInPeriod, earningsDayCountPeriod]

export earningsGoingUp(int _period) =>
    totalShares1M = syminfo.shares_outstanding_total / 1000000.0
    epsAct1M = request.earnings(ticker = syminfo.tickerid, field = earnings.actual) / totalShares1M
    epsStd1M = request.earnings(ticker = syminfo.tickerid, field = earnings.standardized) / totalShares1M
    [epsBetterInPeriod, earningsDayCountPeriod] = earningsGoingUp(epsAct1M, epsStd1M, _period)

export isMinevriniBullishTrend(int _tradingDaysInAYear, bool _includeRelativeStrength, bool _includeEarnings, float _minEarningsChangeRate) =>
    priceMasBulishAlignedCond = priceMasBullishAligned(_tradingDaysInAYear)
    simpleRelativeStrengthCond = simpleRelativeStrengthStrong(_tradingDaysInAYear) and _includeRelativeStrength or not _includeRelativeStrength
    [epsBetter1Y, epsDays1Y] = earningsGoingUp(_tradingDaysInAYear + 20)
    earningsAvailable = epsDays1Y > 0
    epsGoingUpCond = epsBetter1Y / epsDays1Y >= _minEarningsChangeRate
    earningsCond = epsGoingUpCond and _includeEarnings or not _includeEarnings or not earningsAvailable
    bullishTrend = priceMasBulishAlignedCond and simpleRelativeStrengthCond and earningsCond
    bullishTrend

export getWideIndexAndSectorSignalMod(int daysSinceTrend) =>
    wideIndexClose = getDefaultBenchmark(close)
    [wideIndexBullish, wideIndexBearish] = priceMasTrendAligned(wideIndexClose)
    isWideIndexRecentlyBullishCond = nz(ta.barssince(wideIndexBullish), 9999) < daysSinceTrend
    isWideIndexRecentlyBearishCond = nz(ta.barssince(wideIndexBearish), 9999) < daysSinceTrend
    sectorIndex = getSectorIndex()
    sectorIndexClose = sectorIndex != '' ? request.security(sectorIndex, "D", close) : na
    [sectorIndexBullish, sectorIndexBearish] = priceMasTrendAligned(sectorIndexClose)
    isSectorRecentlyBullishCond = nz(ta.barssince(sectorIndexBullish), 9999) < daysSinceTrend
    isSectorRecentlyBearishCond = nz(ta.barssince(sectorIndexBearish), 9999) < daysSinceTrend
    sigMod = (isWideIndexRecentlyBullishCond ? 1 : 0) - (isWideIndexRecentlyBearishCond ? 1 : 0) + (isSectorRecentlyBullishCond ? 1 : 0) - (isSectorRecentlyBearishCond ? 1 : 0)
    sigMod

export nonBrokenExtremesPct(int _lookback) =>
    int lowsCount = 0
    int highsCount = 0
    float lowSoFar = low
    float highSoFar = high
    for int i = 1 to _lookback
        if (low[i] < lowSoFar)
            lowsCount += 1
            lowSoFar := low[i]
        if (high[i] > highSoFar)
            highsCount += 1
            highSoFar := high[i]
    float nonBrokenLowsPct = 100.0 * lowsCount / _lookback
    float nonBrokenHighsPct = 100.0 * highsCount / _lookback
    float nonBrokenExtremesPct = nonBrokenLowsPct + nonBrokenHighsPct
    float lowsHighsRatio = lowsCount / math.max(highsCount, 0.01)
    [nonBrokenLowsPct, nonBrokenHighsPct, nonBrokenExtremesPct, lowsHighsRatio]

export nonBrokenExtremesPctSigMod(int _lookback, float _highLvl, float _ratioHighLvl) =>
    [nBLowsPct, nBHighsPct, nBBoth, ratio] = nonBrokenExtremesPct(_lookback)
    sigMod = nBBoth > _highLvl ? (ratio > _ratioHighLvl ? 1 : ratio < 1 / _ratioHighLvl ? -1 : 0) : 0
    [nBLowsPct, nBHighsPct, nBBoth, ratio, sigMod]

export nonBrokenExtremesPctSigMod(int _lookback, float _highLvl, float _lowLvl, float _ratioHighLvl) =>
    [nBLowsPct, nBHighsPct, nBBoth, ratio] = nonBrokenExtremesPct(_lookback)
    sigMod = nBBoth > _highLvl ? (ratio > _ratioHighLvl ? 1 : ratio < 1 / _ratioHighLvl ? -1 : 0) : nBBoth < _lowLvl ? -1 : 0
    [nBLowsPct, nBHighsPct, nBBoth, ratio, sigMod]

//Relative Measured Volatility
export rmv(int _mainLookback, int _shortLookback1, int _shortLookback2, int _shortLookback3) =>
    short1 = ta.atr(_shortLookback1)
    short2 = ta.atr(_shortLookback2)
    short3 = ta.atr(_shortLookback3)
    shortAvg = (short1 + short2 + short3) / 3
    highestShortAvg = ta.highest(shortAvg, _mainLookback)
    lowestShortAvg = ta.lowest(shortAvg, _mainLookback)
    rmvValue = (shortAvg - lowestShortAvg) / math.max(highestShortAvg - lowestShortAvg, 0.001) * 100
    rmvValue

export rmvContraction(int _mainLookback, int _shortLookback1, int _shortLookback2, int _shortLookback3, int _contractionLvl) =>
    rmvValue = rmv(_mainLookback, _shortLookback1, _shortLookback2, _shortLookback3)
    contraction = rmvValue < _contractionLvl
    contraction

export donchian(int len) => math.avg(ta.lowest(len), ta.highest(len))

export ichimoku(int _conversionPeriods, int _basePeriods, int _laggingSpan2Periods, int _displacement) =>
    conversionLine = donchian(_conversionPeriods)
    baseLine = donchian(_basePeriods)
    leadLine1 = math.avg(conversionLine, baseLine)[_displacement]
    leadLine2 = donchian(_laggingSpan2Periods)[_displacement]
    [conversionLine, baseLine, leadLine1, leadLine2]

export isAboveIchimokuCloud() =>
    [conversionLine, baseLine, leadLine1, leadLine2] = ichimoku(9, 26, 52, 26)
    ichimokuSigL = close > math.max(leadLine1, leadLine2)
    ichimokuSigS = close < math.min(leadLine1, leadLine2)
    initiallyCouldBeNa = ta.barssince(ichimokuSigL) < ta.barssince(ichimokuSigS)
    aboveIchimokuCloud = ichimokuSigL ? true : ichimokuSigS ? false : initiallyCouldBeNa

//obv
export isAboveSrcSmaStddev(series float _src, int _smaLen, float _stddevMultiple) => 
    srcMa = ta.sma(_src, _smaLen)
    srcStdDev = ta.stdev(_src, _smaLen) * _stddevMultiple
    goL = _src > srcMa + srcStdDev
    goS = _src < srcMa - srcStdDev
    initiallyCouldBeNa = ta.barssince(goL) < ta.barssince(goS)
    aboveObvSmaStddev = goL ? true : goS ? false : initiallyCouldBeNa
    aboveObvSmaStddev

export isAboveObvSmaStddev(int _smaLen, float _stddevMultiple) => 
    aboveObvSmaStddev = isAboveSrcSmaStddev(ta.obv, _smaLen, _stddevMultiple)
    aboveObvSmaStddev

export minTurnoverOk(float _minTurnoverM, int _len) =>
    hasEnoughBars = bar_index >= _len
    minVolume = ta.lowest(volume, _len)
    turnoverMa = ta.sma(close * volume, _len) / 1000000.0
    turnoverMedian = ta.median(close * volume, _len) / 1000000.0
    turnoverCond = (not na(volume) and hasEnoughBars and minVolume > 0 and turnoverMa > _minTurnoverM and turnoverMedian > _minTurnoverM) or na(volume)
    turnoverCond

export minAdrOk(float _minAdr, int _len) =>
    adr = 100 * (ta.sma(high / low, _len) - 1)
    adrCond = adr > _minAdr
    adrCond

export isAboveRecentHighVwap(int _lookback) =>
    localHigh = ta.highest(_lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localHighVwap = ta.vwap(ohlc4, high == localHigh)

export calculateSigModVolumeMultiple(int _totalSigMod, float _positiveSigModVolumeBonus, float _negativeSigModVolumeBonus) => 
    float sigModVolMultiple = 1.0 - (_totalSigMod * (_totalSigMod > 0 ? _positiveSigModVolumeBonus : _totalSigMod < 0 ? _negativeSigModVolumeBonus : 0))
    sigModVolMultiple

export strongBreakout(int _lookback1, bool _lookbackMultiples, int _initialMinVolDttPct, int _dayEndMinVolDttPct, int _sigMod) =>
    lastHighest1 = ta.highest(_lookback1)
    lastHighest1p = ta.highest(_lookback1 + 2)
    lastHighest2 = ta.highest(2 * _lookback1)
    lastHighest2p = ta.highest(2 * _lookback1 + 2)
    lastHighest3 = ta.highest(4 * _lookback1)
    lastHighest4 = ta.highest(7 * _lookback1)
    lastHighest5 = ta.highest(11 * _lookback1)
    breakout1 = close > lastHighest1[1] and close > lastHighest1p[1]
    breakout2 = close > lastHighest2[1] and close > lastHighest2p[1]
    breakout3 = close > lastHighest3[1]
    breakout4 = close > lastHighest4[1]
    breakout5 = close > lastHighest5[1]
    breakout1New = not breakout1[1] and not breakout1[2] and not breakout1[3]
    breakout2New = not breakout2[1] and not breakout2[2] and not breakout2[3]
    breakout3New = not breakout3[1] and not breakout3[2] and not breakout3[3]
    breakout4New = not breakout4[1] and not breakout4[2] and not breakout4[3]
    breakout5New = not breakout5[1] and not breakout5[2] and not breakout5[3]
    breakoutCond = breakout1New or _lookbackMultiples and (breakout2New or breakout3New or breakout4New or breakout5New)
    volumeSigMod = 1 - _sigMod * 0.1
    volumeMa = ta.sma(volume, 30)
    dttVol = volume / volumeMa
    dttTime = dttTimePct()
    dttVolTimePct = 100 * dttVol / dttTime
    //math.pow(1-x, n) gives a nice concave function in x={0, 1} range, the bigger the n the more concave function is
    timeSinceSessionOpenAdditionalVolPct = math.pow(1 - dttTime, 5) * (_initialMinVolDttPct - _dayEndMinVolDttPct) * volumeSigMod
    initialFlat = 0.01
    volumeGood = (dttTime >= initialFlat ? (dttVolTimePct > _dayEndMinVolDttPct + timeSinceSessionOpenAdditionalVolPct) : volume > initialFlat * _initialMinVolDttPct * volumeMa * volumeSigMod) or volume > (_dayEndMinVolDttPct * volumeSigMod / 100) * volumeMa
    strongBreakout = breakout1 and breakoutCond and volumeGood
    strongBreakout

    timeSinceSessionOpenAdditionalVolPct = math.pow(1 - dttTime, 5) * (_initialMinVolDttPct - _dayEndMinVolDttPct) * volumeSigMod
    initialFlat = 0.01

// Signals
export sigParabolicTop(series float _volMa, int _downDaysAfterTop, int _prevPeriodGainLen, int _atrMulti, series float _atr) =>
    barsSinceVolDown = ta.barssince(_volMa < _volMa[3])
    localLow = ta.lowest(close, _prevPeriodGainLen)
    downDaysCond = bar_index > _downDaysAfterTop and volume < volume[_downDaysAfterTop] and open < close[_downDaysAfterTop]
    ret = barsSinceVolDown > 15 and close > localLow + _atrMulti * _atr and downDaysCond
    ret

func_init_vwap_excl_init_higher_volume() =>
    x = 0.5
    reset = bar_index == 0 or bar_index == 1 and volume < x * volume[1] or bar_index == 2 and volume < x * volume[1] and volume[1] < x * volume[2]
    ret = ta.vwap(ohlc4, reset)
    ret

export sigJunior(int _maxAge) =>
    initVwap = func_init_vwap_excl_init_higher_volume()
    ret = bar_index < _maxAge and crossover(initVwap)
    ret

export sigStarter(bool _isBullish, int _maLen, int _minMove1Len, int _maxMove2Len, int _smoothing) =>
    ma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    // Only calculate move1 after MA has enough data
    move1 = bar_index >= _maLen ? math.min(1000, ta.barssince(_isBullish ? ma > ma[1] : ma < ma[1])) : 1000
    move2 = math.min(1000, ta.barssince(move1 == 0 and (_isBullish ? move1[1] > _minMove1Len : move1[1] < _minMove1Len)))
    ret = move2 < _maxMove2Len and (_isBullish ? crossover(ma) and ma >= ma[move2] : crossunder(ma) and ma <= ma[move2])
    ret

export sigFireStarter(int _maLen, int _minMove1Len, float _atrMulti, int _smoothing, float _atr) =>
    sma = _smoothing > 0 ? ta.ema(ta.sma(close, _maLen), _smoothing) : ta.sma(close, _maLen)
    vwma = ta.vwma(ohlc4, _maLen)
    move1 = ta.barssince(sma > sma[1])
    maDiffLen = math.ceil(0.4 * (move1 > _minMove1Len ? move1 : _minMove1Len))
    highestMaDiff = ta.highest(sma - vwma, maDiffLen) / _atr
    ret = move1 > _minMove1Len and (crossover(sma) or crossover(vwma)) and highestMaDiff > _atrMulti and highestMaDiff > highestMaDiff[maDiffLen]
    ret

export sigLonger(series float _shortMa, series float _longMa, float _offset, float _atr) =>
    longMaCond = na(_longMa[1]) ? true : (_longMa > _longMa[1] and _shortMa > _longMa)
    ret = _shortMa > _shortMa[1] and longMaCond and crossover(_shortMa + _offset * _atr)
    ret

export sigBullSLHunter(float _atr, int _lookback, int _gracePeriod) =>
    localLow = ta.lowest(_lookback)
    localLowCandlesAgo = ta.barssince(low == localLow)
    validCandlesAgo = localLowCandlesAgo == 0 ? 1 : nz(localLowCandlesAgo, 1)
    localHigh = ta.highest(math.min(_lookback, validCandlesAgo))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = crossover(localLow[_gracePeriod]) and localHigh[_gracePeriod] > localLow[_gracePeriod] + lohi * _atr and localLow[_gracePeriod] == localLow[2 * _gracePeriod]
    sig

export sigBearSLHunter(float _atr, int _lookback, int _gracePeriod) =>
    localHigh = ta.highest(_lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    validCandlesAgo = nz(localHighCandlesAgo, _lookback)
    localLow = ta.lowest(math.max(1, validCandlesAgo))
    lohi = math.pow(math.log(_lookback), 1.25) - 1.25
    sig = crossunder(localHigh[_gracePeriod]) and localLow[_gracePeriod] < localHigh[_gracePeriod] - lohi * _atr and localHigh[_gracePeriod] == localHigh[2 * _gracePeriod]
    sig

export sigBullStepper(float _atr, series float _mainMa, int _lookback, float _maxMaDist) =>
    localHigh = ta.highest(_lookback)
    localHighCandlesAgo = ta.barssince(high == localHigh)
    localHighVwap = ta.vwap(ohlc4, high == localHigh)
    maDist = localHighVwap - _mainMa
    sig = _mainMa > _mainMa[1] and localHigh[localHighCandlesAgo] > _mainMa[localHighCandlesAgo] + 1.5 * math.log(localHighCandlesAgo) * _atr and maDist > 0 and maDist < _maxMaDist * _atr and crossover(localHighVwap)
    sig
