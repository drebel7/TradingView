//@version=6
indicator('DRE Minevrini Trend BG', overlay = true, timeframe = '1D', max_bars_back = 800)

import drebel/DRELib/4 as drelib

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.2, title = 'Min. Turnover[M]', minval = 0, group = 'General filter for signal')
iMinAdr = input.float(defval = 2.6, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min. price', minval = 0, step = 0.2, group = 'General filter for signal')

//common filters calculations 
gFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, 50) and iGeneralFilterCond or not iGeneralFilterCond

//minevrini trend inputs
iBullish = input.bool(defval = true, title = 'bullish', group = 'trend', inline = 'bb')
iBearish = input.bool(defval = true, title = 'bearish', group = 'trend', inline = 'bb')
iIncludeRelativeStrenth = input.bool(defval = true, title = 'Relative strength', group = 'Include')
iIncludeEarnings = input.bool(defval = true, title = 'Earnings', group = 'Include', inline = 'earnings')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. change rate', group = 'Include', minval = 0.0, maxval = 1.0, step = 0.1, inline = 'earnings')

minevriniTrend = drelib.isMinevriniBullishTrend(iIncludeRelativeStrenth, iIncludeEarnings, iBetterEarningsRate)
minevriniTrendSig = minevriniTrend and not minevriniTrend[1]

minevriniBearishTrend = drelib.isMinevriniBearishTrend(iIncludeRelativeStrenth, iIncludeEarnings, iBetterEarningsRate)
minevriniBearishTrendSig = minevriniBearishTrend and not minevriniBearishTrend[1]


bgcolor(iBullish and minevriniTrend ? color.new(color.green, 85) : na)
bgcolor(iBearish and minevriniBearishTrend ? color.new(color.red, 85) : na)

alertcondition(gFilter and minevriniTrendSig, title = 'Minevrini Trend green', message = 'mTrend - green')
alertcondition(gFilter and minevriniBearishTrendSig, title = 'Minevrini Trend red', message = 'mTrend - red')


iRmvMainLookback = input.int(defval = 21, title = 'RMV main lookback', minval = 10, group = 'Contraction params')
iRmvShortLookback1 = input.int(defval = 8, title = 'RMV short lookback 1', minval = 1, group = 'Contraction params')
iRmvShortLookback2 = input.int(defval = 5, title = 'RMV short lookback 2', minval = 1, group = 'Contraction params')
iRmvShortLookback3 = input.int(defval = 3, title = 'RMV short lookback 3', minval = 1, group = 'Contraction params')
iMaxBarsSinceContraction = input.int(defval = 8, title = 'Max. bars since contraction', minval = 0, group = 'Contraction params')

//Relative Measured Volatility
rmv(int _mainLookback, int _shortLookback1, int _shortLookback2, int _shortLookback3) =>
    short1 = ta.atr(_shortLookback1)
    short2 = ta.atr(_shortLookback2)
    short3 = ta.atr(_shortLookback3)
    shortAvg = (short1 + short2 + short3) / 3
    highestShortAvg = ta.highest(shortAvg, _mainLookback)
    lowestShortAvg = ta.lowest(shortAvg, _mainLookback)
    rmvValue = (shortAvg - lowestShortAvg) / math.max(highestShortAvg - lowestShortAvg, 0.001) * 100
    rmvValue

rmv = rmv(iRmvMainLookback, iRmvShortLookback1, iRmvShortLookback2, iRmvShortLookback3)
contraction1 = rmv < 25
contractionCond = ta.barssince(contraction1) <= iMaxBarsSinceContraction

iBreakoutLookback = input.int(defval = 15, title = 'Breakout lookback (and x2)', minval = 1, group = 'Breakout params')
iInitialMinVolumePct = input.int(defval = 270, title = 'Initial min. relative volume Pct.', minval = 100, group = 'Breakout params')
iDayEndMinVolumePct = input.int(defval = 135, title = 'Day end min. relative volume Pct.', minval = 100, group = 'Breakout params')

strongBreakout(int _highestLookback1, int _highestLookback2, int _initialMinVolDttPct, int _dayEndMinVolDttPct) =>
    lastHighest1 = ta.highest(_highestLookback1)
    lastHighest2 = ta.highest(_highestLookback2)
    breakout = close > lastHighest1[1] or close > lastHighest2[1]
    dttVol = volume / ta.sma(volume, 30)
    dttTime = (math.min(timenow, time_close) - time) / (time_close - time)
    dttVolTimePct = 100 * dttVol / dttTime
    //math.pow(1-x, n) gives a nice concave function in x={0, 1} range, the bigger the n the more concave function is
    timeSinceSessionOpenAdditionalVolPct = math.pow(1 - dttTime, 4) * (_initialMinVolDttPct - _dayEndMinVolDttPct)
    strongBreakout = breakout and dttVolTimePct > _dayEndMinVolDttPct + timeSinceSessionOpenAdditionalVolPct
    strongBreakout

strongBreakoutCond = strongBreakout(iBreakoutLookback, 2 * iBreakoutLookback, iInitialMinVolumePct, iDayEndMinVolumePct)

minevriniTrendCond = ta.barssince(minevriniTrend) < 50

minevriniTrendContractionBreakout = minevriniTrendCond and contractionCond and strongBreakoutCond

alertcondition(gFilter and minevriniTrendContractionBreakout, title = 'bullish MTCB', message = 'MTCB')
plotshape(minevriniTrendContractionBreakout, title = 'mtcb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtcb', textcolor = color.new(color.green, 0))
