indicator('DRE Minevrini Trend BG', overlay = true, max_bars_back = 800)

import drebel/DRELib/34 as drelib
import drebel/DREPerf/11 as dreperf

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.25, title = 'Min. Turnover[M]', minval = 0.2, group = 'General filter for signal')
iMinAdr = input.float(defval = 2.6, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min.-max. price', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxPrice = input.float(defval = 1750.0, title = '-', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxDistFromLoDPct = input.int(defval = 67, title = 'Max. distance from LoD [%]', minval = 0, maxval = 300, step = 5, group = 'General filter for signal')
iMinBenchmarkNotBearishSince = input.int(defval = 31, title = 'Min. days from benchmark bearish', minval = 0, maxval = 63, group = 'General filter for signal')

//common filters calculations 
int sigMod = drelib.getWideIndexAndSectorSignalMod(iMinBenchmarkNotBearishSince)
bool generalFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, iMaxPrice, 30, iMaxDistFromLoDPct, sigMod)
bool gFilter = generalFilter and iGeneralFilterCond or not iGeneralFilterCond

//minevrini trend inputs
iBullish = input.bool(defval = true, title = 'bullish', group = 'trend', inline = 'bb')
iBearish = input.bool(defval = true, title = 'bearish', group = 'trend', inline = 'bb')
iIncludeRelativeStrenth = input.bool(defval = true, title = 'Relative strength', group = 'Include')
iIncludeEarnings = input.bool(defval = true, title = 'Earnings', group = 'Include', inline = 'earnings')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. change rate', group = 'Include', minval = 0.0, maxval = 1.0, step = 0.1, inline = 'earnings')

doIncludeEarnings = sigMod >= 1 ? false : iIncludeEarnings
minevriniTrend = drelib.isMinevriniBullishTrend(iIncludeRelativeStrenth, doIncludeEarnings, iBetterEarningsRate)
minevriniTrendSig = minevriniTrend and not minevriniTrend[1]

minevriniBearishTrend = drelib.isMinevriniBearishTrend(iIncludeRelativeStrenth, doIncludeEarnings, iBetterEarningsRate)
minevriniBearishTrendSig = minevriniBearishTrend and not minevriniBearishTrend[1]


bgcolor(iBullish and minevriniTrend ? color.new(color.green, 85) : na)
bgcolor(iBearish and minevriniBearishTrend ? color.new(color.red, 85) : na)

alertcondition(gFilter and minevriniTrendSig, title = 'Minevrini Trend green', message = 'mTrend - green')
alertcondition(gFilter and minevriniBearishTrendSig, title = 'Minevrini Trend red', message = 'mTrend - red')

iRmvMainLookback = input.int(defval = 13, title = 'RMV main lookback', minval = 10, group = 'Contraction params')
iRmvShortLookback1 = input.int(defval = 8, title = 'RMV short lookbacks', minval = 1, inline = 'rmv s. lookback', group = 'Contraction params')
iRmvShortLookback2 = input.int(defval = 5, title = '', minval = 1, inline = 'rmv s. lookback', group = 'Contraction params')
iRmvShortLookback3 = input.int(defval = 3, title = '', minval = 1, inline = 'rmv s. lookback', group = 'Contraction params')
iMaxBarsSinceContraction = input.int(defval = 5, title = 'Max. bars since contraction', minval = 0, group = 'Contraction params')

//Relative Measured Volatility
rmvContraction = drelib.rmvContraction(13, 5, 3, 2, 25)
contractionCond0 = ta.barssince(rmvContraction) <= 2
rmvContraction1 = drelib.rmvContraction(13, 8, 5, 3, 25)
contractionCond1 = ta.barssince(rmvContraction1) <= 2
rmvContraction2 = drelib.rmvContraction(16, 8, 5, 3, 25)
contractionCond2 = ta.barssince(rmvContraction2) <= 2
contractionCond3 = ta.barssince(rmvContraction2) <= iMaxBarsSinceContraction
rmvContraction4 = drelib.rmvContraction(21, 8, 5, 3, 25)
contractionCond4 = ta.barssince(rmvContraction4) <= 2
contractionCond5 = ta.barssince(rmvContraction4) <= iMaxBarsSinceContraction
rmvContraction6 = drelib.rmvContraction(30, 13, 8, 5, 25)
contractionCond6 = ta.barssince(rmvContraction6) <= 5
rmvContraction7 = drelib.rmvContraction(55, 34, 21, 13, 25)
contractionCond7 = ta.barssince(rmvContraction7) <= 2
rmvContraction8 = drelib.rmvContraction(55, 34, 21, 13, 25)
contractionCond8 = ta.barssince(rmvContraction8) <= 5

iBreakoutLookback = input.int(defval = 7, title = 'Min. breakout lookback', minval = 1, maxval = 21, group = 'Breakout params')
iBreakoutMultiples = input.bool(defval = true, title = 'Use also multiples of lookback', group = 'Breakout params')
iInitialMinVolumePct = input.int(defval = 600, title = 'Initial min. relative volume Pct.', minval = 100, group = 'Breakout params')
iDayEndMinVolumePct = input.int(defval = 128, title = 'Day end min. relative volume Pct.', minval = 100, group = 'Breakout params')
iSigModVolumePositiveBonus = input.float(defval = 0.1, title = "SigMod Volume Positive Bonus", minval = 0.0, maxval = 0.5, step = 0.01, group = 'Breakout params')
iSigModVolumeNegativeBonus = input.float(defval = 0.15, title = "SigMod Volume Negative Bonus", minval = 0.0, maxval = 0.5, step = 0.01, group = 'Breakout params')

float sigModVolMultiple = drelib.calculateSigModVolumeMultiple(sigMod, iSigModVolumePositiveBonus, iSigModVolumeNegativeBonus)
strongBreakoutCond = drelib.strongBreakout(iBreakoutLookback, iBreakoutMultiples, iInitialMinVolumePct, iDayEndMinVolumePct, sigModVolMultiple, true)

minevriniTrendCond = ta.barssince(minevriniTrend) < 63

mtsb = minevriniTrendCond and strongBreakoutCond
mtcb = minevriniTrendCond and contractionCond0 and strongBreakoutCond
mtcb1 = minevriniTrendCond and contractionCond1 and strongBreakoutCond
mtcb2 = minevriniTrendCond and contractionCond2 and strongBreakoutCond
mtcb3 = minevriniTrendCond and contractionCond3 and strongBreakoutCond
mtcb4 = minevriniTrendCond and contractionCond4 and strongBreakoutCond
mtcb5 = minevriniTrendCond and contractionCond5 and strongBreakoutCond
mtcb6 = minevriniTrendCond and contractionCond6 and strongBreakoutCond
mtcb7 = minevriniTrendCond and contractionCond7 and strongBreakoutCond
mtcb8 = minevriniTrendCond and contractionCond8 and strongBreakoutCond

//additionali signal if there is an OpeningRangeBrakeout the next day after main signal
orb = drelib.openingRangeBreakout(5, true, true)
mtsb_p1D = mtsb[1] and orb
mtcb_p1D = mtcb[1] and orb

alertcondition(gFilter and mtsb, title = 'bullish MTSB', message = 'MTSB')
alertcondition(gFilter and (mtsb or mtsb_p1D), title = 'bullish MTSB+', message = 'MTSB+')
alertcondition(gFilter and mtcb, title = 'bullish MTCB', message = 'MTCB')
alertcondition(gFilter and (mtcb or mtcb_p1D), title = 'bullish MTCB+', message = 'MTCB+')
plotshape(mtsb, title = 'mtsb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtsb', textcolor = color.new(color.green, 0))


//check performance
// iShowSigPerfPlots = input.bool(defval = false, title = 'show plots', group = 'Sig. performnce check')
iShowSigPerfTable = input.bool(defval = false, title = 'show table', group = 'Sig. performnce check')
[s1SuccRate, s1SuccCount, s1AllCount, s1RetAvgPct] = dreperf.successRateAllTimeStats(mtsb, mtsb ? low : na)
[s2SuccRate, s2SuccCount, s2AllCount, s2RetAvgPct] = dreperf.successRateAllTimeStats(mtcb, mtcb ? low : na)
[s3SuccRate, s3SuccCount, s3AllCount, s3RetAvgPct] = dreperf.successRateAllTimeStats(mtcb1, mtcb1 ? low : na)
[s4SuccRate, s4SuccCount, s4AllCount, s4RetAvgPct] = dreperf.successRateAllTimeStats(mtcb2, mtcb2 ? low : na)
[s5SuccRate, s5SuccCount, s5AllCount, s5RetAvgPct] = dreperf.successRateAllTimeStats(mtcb3, mtcb3 ? low : na)
[s6SuccRate, s6SuccCount, s6AllCount, s6RetAvgPct] = dreperf.successRateAllTimeStats(mtcb4, mtcb4 ? low : na)
[s7SuccRate, s7SuccCount, s7AllCount, s7RetAvgPct] = dreperf.successRateAllTimeStats(mtcb5, mtcb5 ? low : na)
[s8SuccRate, s8SuccCount, s8AllCount, s8RetAvgPct] = dreperf.successRateAllTimeStats(mtcb6, mtcb6 ? low : na)
[s9SuccRate, s9SuccCount, s9AllCount, s9RetAvgPct] = dreperf.successRateAllTimeStats(mtcb7, mtcb7 ? low : na)
[s10SuccRate, s10SuccCount, s10AllCount, s10RetAvgPct] = dreperf.successRateAllTimeStats(mtcb8, mtcb8 ? low : na)

//sig stats table
adrPct = 100.0 * (ta.sma(high / low, 30) - 1)
adrCond = 0.8 * adrPct
bool s1SuccRateOk = s1SuccRate > 39 and s1RetAvgPct > adrCond 
bool s2SuccRateOk = s2SuccRate > 39 and s2RetAvgPct > adrCond 
bool s3SuccRateOk = s3SuccRate > 39 and s3RetAvgPct > adrCond 
bool s4SuccRateOk = s4SuccRate > 39 and s4RetAvgPct > adrCond 
bool s5SuccRateOk = s5SuccRate > 39 and s5RetAvgPct > adrCond 
bool s6SuccRateOk = s6SuccRate > 39 and s6RetAvgPct > adrCond 
bool s7SuccRateOk = s7SuccRate > 39 and s7RetAvgPct > adrCond 
bool s8SuccRateOk = s8SuccRate > 39 and s8RetAvgPct > adrCond 
bool s9SuccRateOk = s9SuccRate > 39 and s9RetAvgPct > adrCond 
bool s10SuccRateOk = s10SuccRate > 39 and s10RetAvgPct > adrCond

s1SuccRateStr = dreperf.formatSuccessRateStr(true, s1SuccRateOk, s1SuccRate, s1SuccCount, s1AllCount, s1RetAvgPct)
s2SuccRateStr = dreperf.formatSuccessRateStr(true, s2SuccRateOk, s2SuccRate, s2SuccCount, s2AllCount, s2RetAvgPct)
s3SuccRateStr = dreperf.formatSuccessRateStr(true, s3SuccRateOk, s3SuccRate, s3SuccCount, s3AllCount, s3RetAvgPct)
s4SuccRateStr = dreperf.formatSuccessRateStr(true, s4SuccRateOk, s4SuccRate, s4SuccCount, s4AllCount, s4RetAvgPct)
s5SuccRateStr = dreperf.formatSuccessRateStr(true, s5SuccRateOk, s5SuccRate, s5SuccCount, s5AllCount, s5RetAvgPct)
s6SuccRateStr = dreperf.formatSuccessRateStr(true, s6SuccRateOk, s6SuccRate, s6SuccCount, s6AllCount, s6RetAvgPct)
s7SuccRateStr = dreperf.formatSuccessRateStr(true, s7SuccRateOk, s7SuccRate, s7SuccCount, s7AllCount, s7RetAvgPct)
s8SuccRateStr = dreperf.formatSuccessRateStr(true, s8SuccRateOk, s8SuccRate, s8SuccCount, s8AllCount, s8RetAvgPct)
s9SuccRateStr = dreperf.formatSuccessRateStr(true, s9SuccRateOk, s9SuccRate, s9SuccCount, s9AllCount, s9RetAvgPct)
s10SuccRateStr = dreperf.formatSuccessRateStr(true, s10SuccRateOk, s10SuccRate, s10SuccCount, s10AllCount, s10RetAvgPct)

var tbl = table.new(position = position.top_right, columns = 11, rows = 11)
if barstate.islast and iShowSigPerfTable
    table.cell(table_id = tbl, column = 0, row = 0, text = "sig")
    table.cell(table_id = tbl, column = 1, row = 0, text = "perf")

    table.cell(table_id = tbl, column = 0, row = 1, text = "mtsb")
    table.cell(table_id = tbl, column = 1, row = 1, text = s1SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 2, text = "mtcb")
    table.cell(table_id = tbl, column = 1, row = 2, text = s2SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 3, text = "mtcb1")
    table.cell(table_id = tbl, column = 1, row = 3, text = s3SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 4, text = "mtcb2")
    table.cell(table_id = tbl, column = 1, row = 4, text = s4SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 5, text = "mtcb3.")
    table.cell(table_id = tbl, column = 1, row = 5, text = s5SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 6, text = "mtcb4")
    table.cell(table_id = tbl, column = 1, row = 6, text = s6SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 7, text = "mtcb5.")
    table.cell(table_id = tbl, column = 1, row = 7, text = s7SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 8, text = "mtcb6")
    table.cell(table_id = tbl, column = 1, row = 8, text = s8SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 9, text = "mtcb7")
    table.cell(table_id = tbl, column = 1, row = 9, text = s9SuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 10, text = "mtcb8.")
    table.cell(table_id = tbl, column = 1, row = 10, text = s10SuccRateStr)
