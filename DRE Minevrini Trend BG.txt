//@version=6
indicator('DRE Minevrini Trend BG', overlay = true, timeframe = '1D', max_bars_back = 800)

import drebel/DRELib/3 as drelib

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.2, title = 'Min. Turnover[M]', minval = 0, group = 'General filter for signal')
iMinAdr = input.float(defval = 2.6, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min. price', minval = 0, step = 0.2, group = 'General filter for signal')

//common filters calculations 
gFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, 50) and iGeneralFilterCond or not iGeneralFilterCond

//minevrini trend inputs
iBullish = input.bool(defval = true, title = 'bullish', group = 'trend', inline = 'bb')
iBearish = input.bool(defval = true, title = 'bearish', group = 'trend', inline = 'bb')
iIncludeRelativeStrenth = input.bool(defval = true, title = 'Relative strength', group = 'Include')
iIncludeEarnings = input.bool(defval = true, title = 'Earnings', group = 'Include', inline = 'earnings')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. change rate', group = 'Include', minval = 0.0, maxval = 1.0, step = 0.1, inline = 'earnings')

// substitute for "RS Score > 70"
simpleRelativeStrengthStrong() =>
    benchmark = syminfo.prefix == 'GPW' ? request.security('GPW:WIG', timeframe.period, close) : request.security('SP:SPX', timeframe.period, close)
    relativelyStrong = close / benchmark > close[126] / benchmark[126] and close / benchmark > close[63] / benchmark[63]
    relativelyStrong

simpleRelativeStrengthWeak() =>
    benchmark = syminfo.prefix == 'GPW' ? request.security('GPW:WIG', timeframe.period, close) : request.security('SP:SPX', timeframe.period, close)
    relativelyWeak = close / benchmark < close[126] / benchmark[126] and close / benchmark < close[63] / benchmark[63]
    relativelyWeak

isMinevriniBullishTrend(int _tradingDaysInAYear, bool _includeRelativeStrength, bool _includeEarnings, float _minEarningsChangeRate) =>
    priceMasBulishAlignedCond = drelib.priceMasBullishAligned(_tradingDaysInAYear)
    simpleRelativeStrengthCond = simpleRelativeStrengthStrong() and _includeRelativeStrength or not _includeRelativeStrength
    [epsBetter1Y, epsDays1Y] = drelib.earningsGoingUp(_tradingDaysInAYear + 20)
    earningsAvailable = epsDays1Y > 0
    epsGoingUpCond = epsBetter1Y / epsDays1Y >= _minEarningsChangeRate
    earningsCond = epsGoingUpCond and _includeEarnings or not _includeEarnings or not earningsAvailable
    bullishTrend = priceMasBulishAlignedCond and simpleRelativeStrengthCond and earningsCond
    bullishTrend

isMinevriniBearishTrend(int _tradingDaysInAYear, bool _includeRelativeStrength, bool _includeEarnings, float _minEarningsChangeRate) =>
    priceMasBearishAlignedCond = drelib.priceMasBearishAligned(_tradingDaysInAYear)
    simpleRelativeStrengthCond = simpleRelativeStrengthWeak() and _includeRelativeStrength or not _includeRelativeStrength
    [epsBetter1Y, epsDays1Y] = drelib.earningsGoingUp(_tradingDaysInAYear + 20)
    earningsAvailable = epsDays1Y > 0
    epsGoingUpCond = (epsDays1Y - epsBetter1Y) / epsDays1Y >= _minEarningsChangeRate
    earningsCond = epsGoingUpCond and _includeEarnings or not _includeEarnings or not earningsAvailable
    bearishTrend = priceMasBearishAlignedCond and simpleRelativeStrengthCond and earningsCond
    bearishTrend

minevriniTrend = isMinevriniBullishTrend(252, iIncludeRelativeStrenth, iIncludeEarnings, iBetterEarningsRate)
minevriniTrendSig = minevriniTrend and not minevriniTrend[1]

minevriniBearishTrend = isMinevriniBearishTrend(252, iIncludeRelativeStrenth, iIncludeEarnings, iBetterEarningsRate)
minevriniBearishTrendSig = minevriniBearishTrend and not minevriniBearishTrend[1]


bgcolor(iBullish and minevriniTrend ? color.new(color.green, 85) : na)
bgcolor(iBearish and minevriniBearishTrend ? color.new(color.red, 85) : na)

alertcondition(gFilter and minevriniTrendSig, title = 'Minevrini Trend green', message = 'mTrend - green')
alertcondition(gFilter and minevriniBearishTrendSig, title = 'Minevrini Trend red', message = 'mTrend - red')
