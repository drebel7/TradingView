//@version=6
indicator('DRE Minevrini Trend BG', overlay = true, max_bars_back = 800)

import drebel/DRELib/35 as drelib
import drebel/DREPerf/17 as dreperf

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.25, title = 'Min. Turnover[M]', minval = 0.2, group = 'General filter for signal')
iMinAdr = input.float(defval = 2.6, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min.-max. price', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxPrice = input.float(defval = 1750.0, title = '-', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxDistFromLoDPct = input.int(defval = 67, title = 'Max. distance from LoD [%]', minval = 0, maxval = 300, step = 5, group = 'General filter for signal')
iMinBenchmarkNotBearishSince = input.int(defval = 31, title = 'Min. days from benchmark bearish', minval = 0, maxval = 63, group = 'General filter for signal')

//common filters calculations 
int sigMod = drelib.getWideIndexAndSectorSignalMod(iMinBenchmarkNotBearishSince)
bool generalFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, iMaxPrice, 30, iMaxDistFromLoDPct, sigMod)
bool gFilter = generalFilter and iGeneralFilterCond or not iGeneralFilterCond

//sig success rate filter
iShowMtsbSig = input.bool(defval = true, title = 'Show mtsb', group = 'Signals')
iShowMtc1sbSig = input.bool(defval = true, title = 'Show mtc1sb', group = 'Signals')
iShowMtc2sbSig = input.bool(defval = true, title = 'Show mtc2sb', group = 'Signals')
iShowMtc3sbSig = input.bool(defval = true, title = 'Show mtc3sb', group = 'Signals')
iMinSigSuccessRate = input.float(defval = 39.0, title = 'Min. sig success rate pct.', minval = 0.0, maxval = 100.0, step = 1.0, group = 'Signals')
iMinSigReturnAdrMultiple = input.float(defval = 0.85, title = 'Min. avg. return by ADR multiple', minval = -1.0, maxval = 5.0, step = 0.05, group = 'Signals')

//minevrini trend inputs
iBullish = input.bool(defval = true, title = 'bullish', group = 'trend', inline = 'bb')
iBearish = input.bool(defval = true, title = 'bearish', group = 'trend', inline = 'bb')
iIncludeRelativeStrenth = input.bool(defval = true, title = 'Relative strength', group = 'Include')
iIncludeEarnings = input.bool(defval = true, title = 'Earnings', group = 'Include', inline = 'earnings')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. change rate', group = 'Include', minval = 0.0, maxval = 1.0, step = 0.1, inline = 'earnings')

doIncludeEarnings = sigMod >= 1 ? false : iIncludeEarnings
minevriniTrend = drelib.isMinevriniBullishTrend(iIncludeRelativeStrenth, doIncludeEarnings, iBetterEarningsRate)
minevriniTrendSig = minevriniTrend and not minevriniTrend[1]

minevriniBearishTrend = drelib.isMinevriniBearishTrend(iIncludeRelativeStrenth, doIncludeEarnings, iBetterEarningsRate)
minevriniBearishTrendSig = minevriniBearishTrend and not minevriniBearishTrend[1]


bgcolor(iBullish and minevriniTrend ? color.new(color.green, 85) : na)
bgcolor(iBearish and minevriniBearishTrend ? color.new(color.red, 85) : na)

alertcondition(gFilter and minevriniTrendSig, title = 'Minevrini Trend green', message = 'mTrend - green')
alertcondition(gFilter and minevriniBearishTrendSig, title = 'Minevrini Trend red', message = 'mTrend - red')

iContr1RmvMainLookback = input.int(defval = 13, title = 'Lookbacks (main/s1/s2/s3)', minval = 8, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS1Lookback = input.int(defval = 5, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS2Lookback = input.int(defval = 3, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS3Lookback = input.int(defval = 2, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1MaxBarsSinceContraction = input.int(defval = 2, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 1 (v. short)')

iContr2RmvMainLookback = input.int(defval = 21, title = 'Lookbacks (main/s1/s2/s3)', minval = 10, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS1Lookback = input.int(defval = 8, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS2Lookback = input.int(defval = 5, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS3Lookback = input.int(defval = 3, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2MaxBarsSinceContraction = input.int(defval = 3, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 2 (short)')

iContr3RmvMainLookback = input.int(defval = 55, title = 'Lookbacks (main/s1/s2/s3)', minval = 10, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS1Lookback = input.int(defval = 34, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS2Lookback = input.int(defval = 21, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS3Lookback = input.int(defval = 13, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3MaxBarsSinceContraction = input.int(defval = 5, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 3 (medium)')

//Relative Measured Volatility
rmvContraction1 = drelib.rmvContraction(iContr1RmvMainLookback, iContr1RmvS1Lookback, iContr1RmvS2Lookback, iContr1RmvS3Lookback, 25)
contractionCond1 = ta.barssince(rmvContraction1) <= iContr1MaxBarsSinceContraction
rmvContraction2 = drelib.rmvContraction(iContr2RmvMainLookback, iContr2RmvS1Lookback, iContr2RmvS2Lookback, iContr2RmvS3Lookback, 25)
contractionCond2 = ta.barssince(rmvContraction2) <= iContr2MaxBarsSinceContraction
rmvContraction3 = drelib.rmvContraction(iContr3RmvMainLookback, iContr3RmvS1Lookback, iContr3RmvS2Lookback, iContr3RmvS3Lookback, 25)
contractionCond3 = ta.barssince(rmvContraction3) <= iContr3MaxBarsSinceContraction

iBreakoutLookback = input.int(defval = 7, title = 'Min. breakout lookback', minval = 1, maxval = 21, group = 'Breakout params')
iBreakoutMultiples = input.bool(defval = true, title = 'Use also multiples of lookback', group = 'Breakout params')
iInitialMinVolumePct = input.int(defval = 600, title = 'Initial min. relative volume Pct.', minval = 100, group = 'Breakout params')
iDayEndMinVolumePct = input.int(defval = 128, title = 'Day end min. relative volume Pct.', minval = 100, group = 'Breakout params')
iSigModVolumePositiveBonus = input.float(defval = 0.1, title = "SigMod Volume Positive Bonus", minval = 0.0, maxval = 0.5, step = 0.01, group = 'Breakout params')
iSigModVolumeNegativeBonus = input.float(defval = 0.15, title = "SigMod Volume Negative Bonus", minval = 0.0, maxval = 0.5, step = 0.01, group = 'Breakout params')

float sigModVolMultiple = drelib.calculateSigModVolumeMultiple(sigMod, iSigModVolumePositiveBonus, iSigModVolumeNegativeBonus)
strongBreakoutCond = drelib.strongBreakout(iBreakoutLookback, iBreakoutMultiples, iInitialMinVolumePct, iDayEndMinVolumePct, sigModVolMultiple, true)

minevriniTrendCond = ta.barssince(minevriniTrend) < 63

mtsb = minevriniTrendCond and strongBreakoutCond
mtc1sb = minevriniTrendCond and contractionCond1 and strongBreakoutCond
mtc2sb = minevriniTrendCond and contractionCond2 and strongBreakoutCond
mtc3sb = minevriniTrendCond and contractionCond3 and strongBreakoutCond

//additionali signal if there is an OpeningRangeBrakeout the next day after main signal
orb = drelib.openingRangeBreakout(5, true, true)
mtsb_p1D = mtsb[1] and orb
mtc1sb_p1D = mtc1sb[1] and orb
mtc2sb_p1D = mtc2sb[1] and orb
mtc3sb_p1D = mtc3sb[1] and orb

alertcondition(gFilter and mtsb, title = 'bullish MTSB', message = 'MTSB')
alertcondition(gFilter and (mtsb or mtsb_p1D), title = 'bullish MTSB+', message = 'MTSB+')
alertcondition(gFilter and mtc1sb, title = 'bullish MTC1B', message = 'MTC1B')
alertcondition(gFilter and (mtc1sb or mtc1sb_p1D), title = 'bullish MTC1B+', message = 'MTC1B+')
alertcondition(gFilter and mtc2sb, title = 'bullish MTC2B', message = 'MTC2B')
alertcondition(gFilter and (mtc2sb or mtc2sb_p1D), title = 'bullish MTC2B+', message = 'MTC2B+')
alertcondition(gFilter and mtc3sb, title = 'bullish MTC3B', message = 'MTC3B')
alertcondition(gFilter and (mtc3sb or mtc3sb_p1D), title = 'bullish MTC3B+', message = 'MTC3B+')
plotshape(iShowMtsbSig and mtsb, title = 'mtsb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtsb', textcolor = color.new(color.green, 0))
plotshape(iShowMtc1sbSig and mtc1sb, title = 'mtc1sb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtc1sb', textcolor = color.new(color.green, 0))
plotshape(iShowMtc2sbSig and mtc2sb, title = 'mtc2sb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtc2sb', textcolor = color.new(color.green, 0))
plotshape(iShowMtc3sbSig and mtc3sb, title = 'mtc3sb', location = location.belowbar, color = color.new(color.green, 0), style = shape.triangleup, text = 'mtc3sb', textcolor = color.new(color.green, 0))


//check performance
//sig stats table
adrPct = 100.0 * (ta.sma(high / low, 100) - 1)
adrCond = iMinSigReturnAdrMultiple * adrPct

// iShowSigPerfPlots = input.bool(defval = false, title = 'show plots', group = 'Sig. performnce check')
iShowSigPerfTable = input.bool(defval = false, title = 'show table', group = 'Sig. performnce check')
[mtsbSuccRate, mtsbSuccRateOk, mtsbSuccRateStr] = dreperf.successRateAllTimeStatsMin(mtsb, mtsb ? low : na, iMinSigSuccessRate, adrCond)
[mtc1sbSuccRate, mtc1sbSuccRateOk, mtc1sbSuccRateStr] = dreperf.successRateAllTimeStatsMin(mtc1sb, mtc1sb ? low : na, iMinSigSuccessRate, adrCond)
[mtc2sbSuccRate, mtc2sbSuccRateOk, mtc2sbSuccRateStr] = dreperf.successRateAllTimeStatsMin(mtc2sb, mtc2sb ? low : na, iMinSigSuccessRate, adrCond)
[mtc3sbSuccRate, mtc3sbSuccRateOk, mtc3sbSuccRateStr] = dreperf.successRateAllTimeStatsMin(mtc3sb, mtc3sb ? low : na, iMinSigSuccessRate, adrCond)

var tbl = table.new(position = position.top_right, columns = 2, rows = 5)
if barstate.islast and iShowSigPerfTable
    table.cell(table_id = tbl, column = 0, row = 0, text = "sig")
    table.cell(table_id = tbl, column = 1, row = 0, text = "perf")

    table.cell(table_id = tbl, column = 0, row = 1, text = "mtsb")
    table.cell(table_id = tbl, column = 1, row = 1, text = mtsbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 2, text = "mtc1sb")
    table.cell(table_id = tbl, column = 1, row = 2, text = mtc1sbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 3, text = "mtc2sb")
    table.cell(table_id = tbl, column = 1, row = 3, text = mtc2sbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 4, text = "mtc3sb")
    table.cell(table_id = tbl, column = 1, row = 4, text = mtc3sbSuccRateStr)
