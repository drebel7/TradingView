//@version=6
indicator('DRE Minevrini Trend BG', overlay = true, timeframe = '1D', max_bars_back = 800)

//common filters inputs
iTurnoverAndAdrCond = input.bool(defval = true, title = 'Min. Turnover and ADR for signal [M]', group = 'Signal filters', inline = 'trn')
iMinTurnover = input.float(defval = 0.2, title = '', minval = 0, group = 'Signal filters', inline = 'trn')
iMinAdr = input.float(defval = 2.6, title = '', minval = 0, step = 0.2, group = 'Signal filters', inline = 'adr')
iSigDelay = input.int(defval = 0, title = 'Max. delay (days)', group = 'Signal filters', minval = 0, maxval = 60)

//common filters calculations 
minTurnoverOk(_minTurnoverM, _len) =>
    turnoverMa = ta.sma(close * volume, _len) / 1000000.0
    turnoverMedian = ta.median(close * volume, _len) / 1000000.0
    turnoverCond = turnoverMa > _minTurnoverM and turnoverMedian > _minTurnoverM or not na(volume)
    turnoverCond

minAdrOk(_minAdr, _len) =>
    adr = 100 * (ta.sma(high / low, _len) - 1)
    adrCond = adr > _minAdr
    adrCond

generalFilter(_minTurnoverM, _minAdr, _minPrice, _len) =>
    turnoverCond = minTurnoverOk(_minTurnoverM, _len)
    adrCond = minAdrOk(_minAdr, _len)
    priceCond = close > _minPrice
    allCond = turnoverCond and adrCond and priceCond
    allCond

gFilter = generalFilter(iMinTurnover, iMinAdr, 1.0, 50) and iTurnoverAndAdrCond or not iTurnoverAndAdrCond

//minevrini trend inputs
iBullish = input.bool(defval = true, title = 'bullish', group = 'trend', inline = 'bb')
iBearish = input.bool(defval = true, title = 'bearish', group = 'trend', inline = 'bb')
iIncludeRelativeStrenth = input.bool(defval = true, title = 'Relative strength', group = 'Include')
iIncludeEarnings = input.bool(defval = true, title = 'Earnings', group = 'Include', inline = 'earnings')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. change rate', group = 'Include', minval = 0.0, maxval = 1.0, step = 0.1, inline = 'earnings')

//minevrini trend calculations
int tradingDaysInAYear = 250

// bullish conditions
priceMasBullishAligned(_tradingDaysInAYear) =>
    ma200 = ta.sma(close, 200)
    ma150 = ta.sma(close, 150)
    ma50 = ta.sma(close, 50)
    yLow = ta.lowest(_tradingDaysInAYear)
    yHigh = ta.highest(_tradingDaysInAYear)
    ma200GoingUp_10 = ma200 > ma200[1] ? 1 : 0
    ma200GoingUpCond = math.sum(ma200GoingUp_10, 35) > 30
    masAlignedCond = ma50 > ma150 and ma50 > ma200 and ma150 > ma200 and ma200GoingUpCond
    priceAlignedCond = close > 1.25 * yLow and close > yHigh / 1.25
    priceMasAlignedCond = close > ma50 / 1.08 and close > ma150 and close > ma200
    allCond = masAlignedCond and priceAlignedCond and priceMasAlignedCond
    allCond

priceMasBulishAligned = priceMasBullishAligned(tradingDaysInAYear)

// substitute for "RS Score > 70"
benchmark = syminfo.prefix == 'GPW' ? request.security('GPW:WIG', timeframe.period, close) : request.security('SP:SPX', timeframe.period, close)
simpleRelativeStrengthCond = iIncludeRelativeStrenth and close / benchmark > close[126] / benchmark[126] and close / benchmark > close[63] / benchmark[63] or not iIncludeRelativeStrenth

// earnings acceleratiion
totalShares = syminfo.shares_outstanding_total
epsAct = request.earnings(ticker = syminfo.tickerid, field = earnings.actual) / totalShares
epsStd = request.earnings(ticker = syminfo.tickerid, field = earnings.standardized) / totalShares
earningsAvailable = not na(epsAct) or not na(epsStd)

earningsGoingUp(_eps, _candlesInAYear) =>
    candlesOf1yearAndSome = _candlesInAYear + 40
    candlesOf2yearsAndSome = 2 * _candlesInAYear + 40
    earningsDay = _eps != _eps[1]
    earningsDayCountIn2Years = math.sum(earningsDay ? 1 : 0, candlesOf2yearsAndSome)
    epsBetter_10 = earningsDay and _eps > _eps[candlesOf1yearAndSome] ? 1 : 0
    epsBetterIn2Years = math.sum(epsBetter_10, candlesOf2yearsAndSome)
    isEnouthEarnings = earningsDayCountIn2Years >= 8
    betterEpsRate = epsBetterIn2Years / earningsDayCountIn2Years
    epsGoingUp = isEnouthEarnings and betterEpsRate >= iBetterEarningsRate and _eps > _eps[candlesOf1yearAndSome]
    epsGoingUp

epsActGoingUp = earningsGoingUp(epsAct, tradingDaysInAYear)
epsStdGoingUp = earningsGoingUp(epsStd, tradingDaysInAYear)
epsGoingUpCond = (epsActGoingUp or epsStdGoingUp) and math.max(nz(epsAct), nz(epsStd)) > 0
earningsCond = iIncludeEarnings and epsGoingUpCond or not iIncludeEarnings or not earningsAvailable

// bearish conditions
earningsGoingDn(_eps, _candlesInAYear) =>
    candlesOf1yearAndSome = _candlesInAYear + 40
    candlesOf2yearsAndSome = 2 * _candlesInAYear + 40
    earningsDay = _eps != _eps[1]
    earningsDayCountIn2Years = math.sum(earningsDay ? 1 : 0, candlesOf2yearsAndSome)
    epsWorse_10 = earningsDay and _eps < _eps[candlesOf1yearAndSome] ? 1 : 0
    epsWorseIn2Years = math.sum(epsWorse_10, candlesOf2yearsAndSome)
    isEnouthEarnings = earningsDayCountIn2Years >= 8
    worseEpsRate = epsWorseIn2Years / earningsDayCountIn2Years
    epsGoingDn = isEnouthEarnings and worseEpsRate >= iBetterEarningsRate and _eps < _eps[candlesOf1yearAndSome]
    epsGoingDn

epsActGoingDn = earningsGoingDn(epsAct, tradingDaysInAYear)
epsStdGoingDn = earningsGoingDn(epsStd, tradingDaysInAYear)
epsGoingDnCond = (epsActGoingDn or epsStdGoingDn) and math.max(nz(epsAct), nz(epsStd)) < 0
earningsDnCond = iIncludeEarnings and epsGoingDnCond or not iIncludeEarnings or not earningsAvailable

priceMasBearishAligned(_tradingDaysInAYear) =>
    ma200 = ta.sma(close, 200)
    ma150 = ta.sma(close, 150)
    ma50 = ta.sma(close, 50)
    yLow = ta.lowest(_tradingDaysInAYear)
    yHigh = ta.highest(_tradingDaysInAYear)
    ma200GoingDn_10 = ma200 < ma200[1] ? 1 : 0
    ma200GoingDnCond = math.sum(ma200GoingDn_10, 35) > 30
    masAlignedCond = ma50 < ma150 and ma50 < ma200 and ma150 < ma200 and ma200GoingDnCond
    priceAlignedCond = close < 1.25 * yLow and close < yHigh / 1.25
    priceMasAlignedCond = close < ma50 * 1.08 and close < ma150 and close < ma200
    allCond = masAlignedCond and priceAlignedCond and priceMasAlignedCond
    allCond

priceMasBearishAligned = priceMasBearishAligned(tradingDaysInAYear)

bearishSimpleRelativeStrengthCond = iIncludeRelativeStrenth and close / benchmark < close[126] / benchmark[126] and close / benchmark < close[63] / benchmark[63] or not iIncludeRelativeStrenth

bearishEarningsCond = iIncludeEarnings and epsGoingDnCond or not iIncludeEarnings or not earningsAvailable


minevriniTrend = priceMasBulishAligned and simpleRelativeStrengthCond and earningsCond
minevriniTrendSig = minevriniTrend and not minevriniTrend[1]

minevriniBearishTrend = priceMasBearishAligned and bearishSimpleRelativeStrengthCond and bearishEarningsCond
minevriniBearishTrendSig = minevriniBearishTrend and not minevriniBearishTrend[1]


bgcolor(iBullish and minevriniTrend ? color.new(color.green, 85) : na)
bgcolor(iBearish and minevriniBearishTrend ? color.new(color.red, 85) : na)

minevriniTrendSigDelayed = ta.barssince(minevriniTrendSig) <= iSigDelay
minevriniBearishTrendSigDelayed = ta.barssince(minevriniBearishTrendSig) <= iSigDelay

alertcondition(gFilter and minevriniTrendSigDelayed, title = 'Minevrini Trend green', message = 'mTrend - green')
alertcondition(gFilter and minevriniBearishTrendSigDelayed, title = 'Minevrini Trend red', message = 'mTrend - red')
