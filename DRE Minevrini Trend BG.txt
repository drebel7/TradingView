//@version=6
indicator("DRE Minevrini Trend BG", overlay = true, max_bars_back = 300)

iBenchmark = input.symbol(defval='GPW:FW201!', title='Benchmark for RS')
iIncldeEarningsIncrease = input.bool(defval=true, title='Include earnings increase')

sma200 = ta.sma(close, 200)
sma150 = ta.sma(close, 150)
sma50 = ta.sma(close, 50)

yealyLow = ta.lowest(250)
yealyHigh = ta.highest(250)

sma200GoingUp_10 = sma200 > sma200[1] ? 1 : 0
sma200GoingUpCond = math.sum(sma200GoingUp_10, 45) > 40
masAlignedCond = sma50 > sma150 and sma50 > sma200 and sma150 > sma200 and sma200GoingUpCond
priceAlignedCond = close > 1.25 * yealyLow and close > yealyHigh / 1.25
priceMasAlignedCond = close > sma50 / 1.08 and close > sma150 and close > sma200

 // and RS Score > 70
benchmark = request.security(iBenchmark, timeframe.period, close)
simpleRelativeStrengthCond = close / benchmark > close[126] / benchmark[126] and close / benchmark > close[63] / benchmark[63]

// earnings ecceleratiion
earnings = request.earnings(ticker = syminfo.tickerid, field = earnings.actual)
totalShares = syminfo.shares_outstanding_total
eps = earnings / totalShares
earningsCond = iIncldeEarningsIncrease and eps > eps[260] or not iIncldeEarningsIncrease

allAligned = masAlignedCond and priceAlignedCond and priceMasAlignedCond and simpleRelativeStrengthCond and earningsCond

bgcolor(allAligned ? color.new(color.green, 85) : na)
