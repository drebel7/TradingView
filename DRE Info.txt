//@version=6
indicator(title = 'DRE Info', shorttitle = 'Info', overlay = true)

import drebel/DRELib/14 as drelib

show_adrp = input.bool(defval = true, title = 'Show ADR %')
show_atr = input.bool(defval = false, title = 'Show ATR')
show_turnover = input.bool(defval = true, title = 'Show avg. turnover')
show_hod_price = input.bool(defval = true, title = 'Show HoD price')
show_lod_price = input.bool(defval = true, title = 'Show LoD price')
show_lod_dist = input.bool(defval = true, title = 'Show LoD distance')
show_dtt_volume = input.bool(defval = true, title = 'Day to time volume %')
show_dtt_time = input.bool(defval = false, title = 'Day to time time %')
show_short_ma_percentile = input.bool(defval = false, title = 'Short MA dist. percentile')
show_mid_ma_percentile = input.bool(defval = false, title = 'Short-Mid. MA dist. percentile')
show_sharpe_ratio = input.bool(defval = false, title = 'Sharpe ratio')
show_earnings_up = input.bool(defval = true, title = 'Earnings going up')
show_industry_sector = input.bool(defval = true, title = 'Industry and Sector')

len = input.int(defval = 50, title = 'Calculations Length', minval = 1, tooltip = 'The number of daily bars used in calculations.')
riskFreeRateAnnual = input.float(defval = 0.05, title = "Risk free annual rate.", minval = 0, step = 0.01)
bg_col = input.color(defval = #00000000, title = 'Background Color')
adrp_col = input.color(defval = color.black, title = 'Text Color')
show_empty_row = input.bool(defval = true, title = 'Add an empty row above the displayed values', tooltip = 'Serves the purpose of not having the pane icons overlap with the ' + 'text in the table if multiple panes are shown and the chart pane is hovered')

hiloAvg = ta.sma(high - low, len)
arp = 100 * (ta.sma(high / low, len) - 1)
[dAdr, dAtr, dAvgTurnover] = request.security(symbol = syminfo.tickerid, timeframe = 'D', expression = [arp, ta.atr(len), ta.sma(close * volume, len) / 1000000])
lodDist = 100 * (close - low) / hiloAvg
dttVolPct = 100 * volume / ta.sma(volume, len)
dttTimePct = 100 * (math.min(timenow, time_close) - time) / (time_close - time)
dttVolTimePct = 100 * dttVolPct / dttTimePct

// Sharpe Ratio
riskFreeRateDaily = riskFreeRateAnnual / 252
logReturn = math.log(close / close[1])
excessReturn = logReturn - riskFreeRateDaily
meanExcessReturn = ta.sma(excessReturn, len)
stdExcessReturn = ta.stdev(excessReturn, len)
sharpeRatio = stdExcessReturn != 0 ? meanExcessReturn / stdExcessReturn * math.sqrt(252) : na

// earnings acceleratiion
tradingDaysInAYear = 252
tradingDaysInAYearAndSome = tradingDaysInAYear + 20
tradingDaysIn2YearsAndSome = 2 * tradingDaysInAYear + 20

[epsBetter1Y, epsDays1Y] = drelib.earningsGoingUp(tradingDaysInAYearAndSome)
[epsBetter2Y, epsDays2Y] = drelib.earningsGoingUp(tradingDaysIn2YearsAndSome)
earningsAvailable = epsDays1Y + epsDays2Y > 0
epsStr = str.tostring(epsBetter1Y) + '/' + str.tostring(epsDays1Y) + ' ' + str.tostring(epsBetter2Y) + '/' + str.tostring(epsDays2Y)

ma2DistPercentile = ta.percentrank(drelib.ma_distance(10, 21), 200)
ma3DistPercentile = ta.percentrank(drelib.ma_distance(10, 21, 50), 200)

sector = syminfo.sector
ind = syminfo.industry

bool isGPW = syminfo.prefix == 'GPW' or syminfo.prefix == 'NEWCONNECT'
string sectorIndex = ''
if isGPW
    if str.contains("NEWCONNECT:AGL,GPW:AGT,GPW:AST,GPW:CLE,GPW:IMC,GPW:KER,GPW:KSG,GPW:MLK", syminfo.tickerid)
        sectorIndex := 'GPW:WIG_UKRAIN'
    else if str.contains("GPW:11B,GPW:3RG,GPW:ART,GPW:BBT,GPW:BCS,GPW:BLO,GPW:CDR,GPW:CIG,GPW:CRJ,GPW:DGE,GPW:GIF,GPW:GOP,GPW:HUG,GPW:MOV,GPW:PCF,GPW:PLW,GPW:RND,GPW:SIM,GPW:TEN,GPW:ULG,GPW:VVD", syminfo.tickerid)
        sectorIndex := 'GPW:WIG_GRY'
    else if syminfo.sector == 'Finance' and (syminfo.industry == 'Regional Banks' or syminfo.industry == 'Major Banks')
        sectorIndex := 'GPW:WIG_BANKI'
    else if syminfo.industry == 'Engineering & Construction' or syminfo.industry == 'Construction Materials' or syminfo.industry == 'Building Products' or syminfo.industry == 'Steel' 
        sectorIndex := 'GPW:WIG_BUDOW'
    else if str.contains(syminfo.industry, 'Chemicals') or str.contains(syminfo.industry, 'Oil Refining')
        sectorIndex := 'GPW:WIG_CHEMIA'
    else if syminfo.industry == 'Electric Utilities' or syminfo.industry == 'Alternative Power Generation'
        sectorIndex := 'GPW:WIG_ENERG'
    else if str.contains(syminfo.industry, 'Minerals') or syminfo.industry == 'Coal'
        sectorIndex := 'GPW:WIG_GORNIC'
    else if syminfo.sector == 'Technology Services' or syminfo.sector == 'Electronic Technology'
        sectorIndex := 'GPW:WIG_INFO'
    else if syminfo.sector == 'Health Technology'
        sectorIndex := 'GPW:WIG_LEKI'
    else if str.contains(syminfo.industry, 'Entertainment') or str.contains(syminfo.industry, 'Marketing') or str.contains(syminfo.industry, 'Magazines') or syminfo.industry == 'Broadcasting'
        sectorIndex := 'GPW:WIG_MEDIA'
    else if str.contains(syminfo.industry, 'Auto Parts') or syminfo.industry == 'Broadcasting' or syminfo.tickerid == 'GPW:APR' or syminfo.tickerid == 'GPW:SNK'
        sectorIndex := 'GPW:WIG_MOTO'
    else if str.contains(syminfo.industry, 'Real Estate') or syminfo.industry == 'Homebuilding'
        sectorIndex := 'GPW:WIG_NRCHOM'
    else if syminfo.sector == 'Consumer Non-Durables' or str.contains(syminfo.industry, 'Aparel/Footwear') or syminfo.industry == 'Textiles'
        sectorIndex := 'GPW:WIG_ODZIEZ'
    else if syminfo.sector == 'Energy Minerals' or syminfo.industry == 'Gas Distributors'
        sectorIndex := 'GPW:WIG_PALIWA'
    else if str.contains(syminfo.industry, 'Food') or str.contains(syminfo.industry, 'Beverages')
        sectorIndex := 'GPW:WIG_SPOZYW'
else
    if syminfo.sector == 'Non-Energy Minerals' and syminfo.industry == 'Precious Metals'
        sectorIndex := 'AMEX:XAU'
    else if syminfo.sector == 'Non-Energy Minerals'
        sectorIndex := 'AMEX:XLB'
    else if syminfo.sector == 'Energy Minerals'
        sectorIndex := 'AMEX:XLE'
    else if syminfo.industry == 'Semiconductors'
        sectorIndex := 'NASDAQ:SOX'
    else if syminfo.sector == 'Electronic Technology' or syminfo.sector == 'Technology Services'
        sectorIndex := 'AMEX:XLK'
    else if syminfo.sector == 'Heath Services' or syminfo.sector == 'Heath Technology'
        sectorIndex := 'AMEX:XBI'
    else if syminfo.sector == 'Consumer Services' or syminfo.sector == 'Consumer Durables' or syminfo.sector == 'Retail Trade'
        sectorIndex := 'AMEX:XLY'
    else if str.contains(syminfo.industry, 'Real Estate') or syminfo.industry == 'Homebuilding'
        sectorIndex := 'AMEX:XLRE'
    else if syminfo.industry == 'Airlines'
        sectorIndex := 'AMEX:JETS'

table t = table.new(position.top_right, 2, 16, bgcolor = bg_col)
txtSize = size.small
if barstate.islast and timeframe.period != 'W' and timeframe.period != 'M'
    if show_empty_row
        table.cell(t, 0, 0, '')
        table.cell(t, 1, 0, '')
    if show_adrp
        table.cell(t, 0, 1, 'ADR [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 1, str.tostring(math.round(dAdr, 1)), text_color = adrp_col, text_size = txtSize)
    if show_atr
        table.cell(t, 0, 2, 'ATR', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 2, str.tostring(math.round(dAtr, 2)), text_color = adrp_col, text_size = txtSize)
    if show_turnover
        table.cell(t, 0, 3, 'Avg. turnover [M]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 3, str.tostring(math.round(dAvgTurnover, 2)), text_color = adrp_col, text_size = txtSize)
    if show_hod_price
        table.cell(t, 0, 4, 'HoD price', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 4, str.tostring(high), text_color = adrp_col, text_size = txtSize)
    if show_lod_price
        table.cell(t, 0, 5, 'LoD price', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 5, str.tostring(low), text_color = adrp_col, text_size = txtSize)
    if show_lod_dist
        table.cell(t, 0, 6, 'LoD dist. [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 6, str.tostring(math.round(lodDist)), text_color = adrp_col, text_size = txtSize)
    if show_dtt_volume
        table.cell(t, 0, 7, 'Dtt vol. [%] (proj.)', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 7, str.tostring(math.round(dttVolPct)) + ' (' + str.tostring(math.round(dttVolTimePct)) + ')', text_color = adrp_col, text_size = txtSize)
    if show_dtt_time
        table.cell(t, 0, 8, 'Dtt time [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 8, str.tostring(math.round(dttTimePct)), text_color = adrp_col, text_size = txtSize)
    if show_short_ma_percentile
        table.cell(t, 0, 9, 'Short MAs prctl.', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 9, str.tostring(math.round(ma2DistPercentile)), text_color = adrp_col, text_size = txtSize)
    if show_mid_ma_percentile
        table.cell(t, 0, 10, 'Short-Mid MAs prctl.', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 10, str.tostring(math.round(ma3DistPercentile)), text_color = adrp_col, text_size = txtSize)
    if show_sharpe_ratio
        table.cell(t, 0, 11, 'Sharpe ratio', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 11, str.tostring(math.round(sharpeRatio, 1)), text_color = adrp_col, text_size = txtSize)
    if show_earnings_up
        table.cell(t, 0, 12, 'EPS up 1y 2y', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 12, epsStr, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 13, 'sector', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 13, sector, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 14, 'industry', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 14, ind, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 15, 'bench. index', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 15, sectorIndex, text_color = adrp_col, text_size = txtSize)
