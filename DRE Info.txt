//@version=6
indicator(title = 'DRE Info', shorttitle = 'Info', overlay = true)

import drebel/DRELib/24 as drelib

show_ticker = input.bool(defval = false, title = 'Show ticker')
show_adrp = input.bool(defval = true, title = 'Show ADR %')
show_atr = input.bool(defval = false, title = 'Show ATR')
show_turnover = input.bool(defval = true, title = 'Show avg. turnover')
show_hod_price = input.bool(defval = true, title = 'Show HoD price')
show_lod_price = input.bool(defval = true, title = 'Show LoD price')
show_lod_dist = input.bool(defval = true, title = 'Show LoD distance')
show_dtt_volume = input.bool(defval = true, title = 'Day to time volume %')
show_dtt_time = input.bool(defval = false, title = 'Day to time time %')
show_ma_percentiles = input.bool(defval = true, title = 'MAs dist. percentiles')
show_sharpe_ratio = input.bool(defval = false, title = 'Sharpe ratio')
show_earnings_up = input.bool(defval = true, title = 'Earnings going up')
show_industry_sector = input.bool(defval = true, title = 'Industry and Sector')
show_non_broken_extremes = input.bool(defval = true, title = 'nBroken extremes')

len = input.int(defval = 50, title = 'Calculations Length', minval = 1, tooltip = 'The number of daily bars used in calculations.')
riskFreeRateAnnual = input.float(defval = 0.05, title = "Risk free annual rate.", minval = 0, step = 0.01)
bg_col = input.color(defval = #00000000, title = 'Background Color')
adrp_col = input.color(defval = color.black, title = 'Text Color')
red_col = input.color(defval = color.red, title = 'Text Color (for sig)')
show_empty_row = input.bool(defval = true, title = 'Add an empty row above the displayed values', tooltip = 'Serves the purpose of not having the pane icons overlap with the ' + 'text in the table if multiple panes are shown and the chart pane is hovered')

hiloAvg = ta.sma(high - low, len)
arp = 100 * (ta.sma(high / low, len) - 1)
[dAdr, dAtr, dAvgTurnover] = request.security(symbol = syminfo.tickerid, timeframe = 'D', expression = [arp, ta.atr(len), ta.sma(close * volume, len) / 1000000])
lodDist = 100 * (close - low) / hiloAvg
dttVolPct = 100 * volume / ta.sma(volume, len)
dttTimePct = 100 * (math.min(timenow, time_close) - time) / (time_close - time)
dttVolTimePct = 100 * dttVolPct / dttTimePct

// Sharpe Ratio
riskFreeRateDaily = riskFreeRateAnnual / 252
logReturn = math.log(close / close[1])
excessReturn = logReturn - riskFreeRateDaily
meanExcessReturn = ta.sma(excessReturn, len)
stdExcessReturn = ta.stdev(excessReturn, len)
sharpeRatio = stdExcessReturn != 0 ? meanExcessReturn / stdExcessReturn * math.sqrt(252) : na

// earnings acceleratiion
tradingDaysInAYear = 252
tradingDaysInAYearAndSome = tradingDaysInAYear + 20
tradingDaysIn2YearsAndSome = 2 * tradingDaysInAYear + 20

[epsBetter1Y, epsDays1Y] = drelib.earningsGoingUp(tradingDaysInAYearAndSome)
[epsBetter2Y, epsDays2Y] = drelib.earningsGoingUp(tradingDaysIn2YearsAndSome)
earningsAvailable = epsDays1Y + epsDays2Y > 0
epsStr = str.tostring(epsBetter1Y) + '/' + str.tostring(epsDays1Y) + ' ' + str.tostring(epsBetter2Y) + '/' + str.tostring(epsDays2Y)

ma1DistPercentile = ta.percentrank(close - ta.sma(close, 10), 500)
ma2DistPercentile = ta.percentrank(close - ta.sma(close, 20), 500)
ma3DistPercentile = ta.percentrank(close - ta.sma(close, 50), 500)
ma4DistPercentile = ta.percentrank(close - ta.sma(close, 200), 500)
masDistPercentileRed = ma1DistPercentile > 90 and ma2DistPercentile > 90 and ma3DistPercentile > 90
masDistPercStr = str.tostring(math.round(ma1DistPercentile)) + '/' + str.tostring(math.round(ma2DistPercentile)) + '/' + str.tostring(math.round(ma3DistPercentile)) + '/' + str.tostring(math.round(ma4DistPercentile))

sector = syminfo.sector
ind = syminfo.industry

string sectorIndex = drelib.getSectorIndex()
mod = drelib.getWideIndexAndSectorSignalMod(22)
modStr = mod == 2 ? '(++)' : mod == 1 ? '(+)' : mod == 0 ? '(0)' : mod == -1 ? '(-)' : '(--)'

[nBLowsPct, nBHighsPct, nBBoth, ratio, mod2] = drelib.nonBrokenExtremesPctSigMod(100, 24, 3.25)
mod2Str = mod2 == 1 ? '(+)' : mod2 == 0 ? '(0)' : '(-)'

table t = table.new(position.top_right, 2, 18, bgcolor = bg_col)
txtSize = size.small
if barstate.islast and timeframe.period != 'W' and timeframe.period != 'M'
    if show_empty_row
        table.cell(t, 0, 0, '')
        table.cell(t, 1, 0, '')
    if show_ticker
        table.cell(t, 0, 1, '')
        table.cell(t, 1, 1, ticker.standard())
    if show_adrp
        table.cell(t, 0, 2, 'ADR [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 2, str.tostring(math.round(dAdr, 1)), text_color = adrp_col, text_size = txtSize)
    if show_atr
        table.cell(t, 0, 3, 'ATR', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 3, str.tostring(math.round(dAtr, 2)), text_color = adrp_col, text_size = txtSize)
    if show_turnover
        table.cell(t, 0, 4, 'Avg. turnover [M]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 4, str.tostring(math.round(dAvgTurnover, 2)), text_color = adrp_col, text_size = txtSize)
    if show_hod_price
        table.cell(t, 0, 5, 'HoD price', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 5, str.tostring(high), text_color = adrp_col, text_size = txtSize)
    if show_lod_price
        table.cell(t, 0, 6, 'LoD price', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 6, str.tostring(low), text_color = adrp_col, text_size = txtSize)
    if show_lod_dist
        table.cell(t, 0, 7, 'LoD dist. [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 7, str.tostring(math.round(lodDist)), text_color = adrp_col, text_size = txtSize)
    if show_dtt_volume
        table.cell(t, 0, 8, 'Dtt vol. [%] (proj.)', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 8, str.tostring(math.round(dttVolPct)) + ' (' + str.tostring(math.round(dttVolTimePct)) + ')', text_color = adrp_col, text_size = txtSize)
    if show_dtt_time
        table.cell(t, 0, 9, 'Dtt time [%]', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 9, str.tostring(math.round(dttTimePct)), text_color = adrp_col, text_size = txtSize)
    if show_ma_percentiles
        table.cell(t, 0, 10, 'Close vs MAs prctl.', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 10, masDistPercStr, text_color = masDistPercentileRed ? red_col : adrp_col, text_size = txtSize)
    if show_sharpe_ratio
        table.cell(t, 0, 11, 'Sharpe ratio', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 11, str.tostring(math.round(sharpeRatio, 1)), text_color = adrp_col, text_size = txtSize)
    if show_earnings_up
        table.cell(t, 0, 12, 'EPS up 1y 2y', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 12, epsStr, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 13, 'sector', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 13, sector, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 14, 'industry', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 14, ind, text_color = adrp_col, text_size = txtSize)
    if show_industry_sector
        table.cell(t, 0, 15, 'bench. index', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 15, sectorIndex + modStr, text_color = adrp_col, text_size = txtSize)
    if show_non_broken_extremes
        table.cell(t, 0, 16, 'nBroken extremes', text_color = adrp_col, text_size = txtSize)
        table.cell(t, 1, 16, str.tostring(math.round(nBLowsPct)) + mod2Str, text_color = adrp_col, text_size = txtSize)
