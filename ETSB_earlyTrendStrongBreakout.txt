//@version=6
indicator("early trend strong breakout", overlay = true, calc_bars_count = 12 * 252)

import drebel/DRELib/35 as drelib
import drebel/DREPerf/22 as dreperf

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.20, title = 'Min. Turnover[M]', minval = 0.2, group = 'General filter for signal')
iMinAdr = input.float(defval = 3, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min.-max. price', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxPrice = input.float(defval = 1750.0, title = '-', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxDistFromLoDPct = input.int(defval = 67, title = 'Max. distance from LoD [%]', minval = 0, maxval = 300, step = 5, group = 'General filter for signal')
iMinBenchmarkNotBearishSince = input.int(defval = 31, title = 'Min. days from benchmark bearish', minval = 0, maxval = 63, group = 'General filter for signal')

//sig success rate filter
iShowEt3sbSig = input.bool(defval = true, title = 'Show et3sb', group = 'Signals')
iShowEt4sbSig = input.bool(defval = true, title = 'Show et4sb', group = 'Signals')
iShowRmvSig = input.bool(defval = true, title = 'Show rmv', group = 'Signals')
iShowRmvPbSig = input.bool(defval = true, title = 'Show rmv-pb', group = 'Signals')
iShowEtcsbSig = input.bool(defval = true, title = 'Show etcsb', group = 'Signals')
iShowObvDivSig = input.bool(defval = true, title = 'Show Obv-Div-sb', group = 'Signals')
iShowMinevriniTrendSig = input.bool(defval = true, title = 'Show MinevriniTrend-contr-sb', group = 'Signals')
iMinSigSuccessRate = input.float(defval = 41.0, title = 'Min. sig success rate pct.', minval = 0.0, maxval = 100.0, step = 1.0, group = 'Signals')
iMinSigReturnAdrMultiple = input.float(defval = 0.9, title = 'Min. avg. return by ADR multiple', minval = -1.0, maxval = 5.0, step = 0.05, group = 'Signals')

//common filters calculations 
int sigMod = drelib.getWideIndexAndSectorSignalMod(iMinBenchmarkNotBearishSince)
bool generalFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, iMaxPrice, 30, iMaxDistFromLoDPct, sigMod)
bool gFilter = generalFilter and iGeneralFilterCond or not iGeneralFilterCond

benchIdx = drelib.getDefaultBenchmarkClose()
rs = close / benchIdx
rsMa = ta.sma(rs, 100)
rsCondRaw = rs > rsMa
rsCond_10 = rsCondRaw ? 1 : 0

obvCondRaw = drelib.isAboveObvSmaStddev(100, 0.35)
obvCond_10 = obvCondRaw ? 1 : 0

iMinNonBrokenLows = input.int(defval = 10, title = 'Min. non-broken Lows', minval = 0, step = 1, maxval = 40, group = 'Other')

[nonBrokenLowsPct, nonBrokenHighsPct, nonBrokenExtremesPct, lowsHighsRatio] = drelib.nonBrokenExtremesPct(100)
nbeCondRaw = nonBrokenLowsPct > nonBrokenHighsPct and nonBrokenExtremesPct >= iMinNonBrokenLows
nbeCond_10 = nbeCondRaw ? 1 : 0

iBreakoutLookback = input.int(defval = 7, title = 'Min. breakout lookback', minval = 1, maxval = 21, group = 'Breakout params')
iBreakoutMultiples = input.bool(defval = true, title = 'Use also multiples of lookback', group = 'Breakout params')
iInitialMinVolumePct = input.int(defval = 600, title = 'Initial min. relative volume Pct.', minval = 100, group = 'Breakout params')
iDayEndMinVolumePct = input.int(defval = 125, title = 'Day end min. relative volume Pct.', minval = 100, group = 'Breakout params')

sigMod += nonBrokenExtremesPct <= 14 ? -1 : 0
strongBreakout = drelib.strongBreakout(iBreakoutLookback, iBreakoutMultiples, iInitialMinVolumePct, iDayEndMinVolumePct, sigMod, true)

aboveIchimokuCloudCondRaw = drelib.isAboveIchimokuCloud()
aboveIchimokuCloudCond_10 = aboveIchimokuCloudCondRaw ? 1 : 0

totalCond = 4
etCondSum = rsCond_10 + obvCond_10 + aboveIchimokuCloudCond_10 + nbeCond_10
et3 = etCondSum == totalCond - 1
et4 = etCondSum == totalCond
et3sb = et3 and strongBreakout
et4sb = et4 and strongBreakout
etsb = et3sb or et4sb

//additionali signal if there is an OpeningRangeBrakeout the next day after main signal
orb = drelib.openingRangeBreakout(5, true, true)

//pct time since session beggining 
dttTime = drelib.dttTimePct()

//rmv sig (like in "DRE parabolic top" v444)
iRmvMinVolumeMultiple = input.float(defval = 1.15, title = 'Rmv volume multiple', minval = 0.0, maxval = 3.0, step = 0.05, group = 'Contraction (RMV)')
iRmvPbLen = input.int(6, title = 'Pullback Len.', display = display.none, group = 'Contraction (RMV)')

iContr1RmvMainLookback = input.int(defval = 13, title = 'Lookbacks (main/s1/s2/s3)', minval = 8, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS1Lookback = input.int(defval = 5, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS2Lookback = input.int(defval = 3, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1RmvS3Lookback = input.int(defval = 2, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 1 (v. short)')
iContr1MaxBarsSinceContraction = input.int(defval = 2, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 1 (v. short)')

iContr2RmvMainLookback = input.int(defval = 21, title = 'Lookbacks (main/s1/s2/s3)', minval = 10, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS1Lookback = input.int(defval = 8, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS2Lookback = input.int(defval = 5, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2RmvS3Lookback = input.int(defval = 3, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 2 (short)')
iContr2MaxBarsSinceContraction = input.int(defval = 3, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 2 (short)')

iContr3RmvMainLookback = input.int(defval = 55, title = 'Lookbacks (main/s1/s2/s3)', minval = 10, maxval = 200, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS1Lookback = input.int(defval = 34, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS2Lookback = input.int(defval = 21, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3RmvS3Lookback = input.int(defval = 13, title = '', minval = 1, maxval = 50, inline = 'rmv lookbacks', group = 'Contraction (RMV) params 3 (medium)')
iContr3MaxBarsSinceContraction = input.int(defval = 5, title = 'Max. bars since contraction', minval = 0, group = 'Contraction (RMV) params 3 (medium)')

shortMa = ta.sma(close, 50)
gVShortMa1 = ta.ema(close, 10)
gVolMa = ta.sma(volume, 30)
rmvLowLevel = 25
otherRmvCond = close > close[2] and volume > iRmvMinVolumeMultiple * gVolMa and close > gVShortMa1 and close > shortMa and dttTime > 0.85

rmvContraction1 = drelib.rmv(iContr1RmvMainLookback, iContr1RmvS1Lookback, iContr1RmvS2Lookback, iContr1RmvS3Lookback)
lowRmv1 = rmvContraction1 <= rmvLowLevel
highRmv1 = rmvContraction1 >= 100 - rmvLowLevel / 2
rmv1Cond = lowRmv1[1] and not lowRmv1 or highRmv1 and lowRmv1[2] or rmvContraction1 == 100 and lowRmv1[3]
bullishRmv1 = rmv1Cond and otherRmvCond

rmvContraction2 = drelib.rmv(iContr2RmvMainLookback, iContr2RmvS1Lookback, iContr2RmvS2Lookback, iContr2RmvS3Lookback)
lowRmv2 = rmvContraction2 <= rmvLowLevel
highRmv2 = rmvContraction2 >= 100 - rmvLowLevel / 2
rmv2Cond = lowRmv2[1] and not lowRmv2 or highRmv2 and lowRmv2[2] or rmvContraction2 == 100 and lowRmv2[3]
bullishRmv2 = rmv2Cond and otherRmvCond

rmvContraction3 = drelib.rmv(iContr3RmvMainLookback, iContr3RmvS1Lookback, iContr3RmvS2Lookback, iContr3RmvS3Lookback)
lowRmv3 = rmvContraction3 <= rmvLowLevel
highRmv3 = rmvContraction3 >= 100 - rmvLowLevel / 2
rmv3Cond = lowRmv3[1] and not lowRmv3 or highRmv3 and lowRmv3[2] or rmvContraction3 == 100 and lowRmv3[3]
bullishRmv3 = rmv3Cond and otherRmvCond

bullishRmv1Ago = ta.barssince(bullishRmv1)
bullishRmv1Pb = bullishRmv1Ago <= iRmvPbLen and bullishRmv1[bullishRmv1Ago] and drelib.crossover(low[bullishRmv1Ago])

bullishRmv2Ago = ta.barssince(bullishRmv2)
bullishRmv2Pb = bullishRmv2Ago <= iRmvPbLen and bullishRmv2[bullishRmv2Ago] and drelib.crossover(low[bullishRmv2Ago])

bullishRmv3Ago = ta.barssince(bullishRmv3)
bullishRmv3Pb = bullishRmv3Ago <= iRmvPbLen and bullishRmv3[bullishRmv3Ago] and drelib.crossover(low[bullishRmv3Ago])


etc1sb = etsb and (rmv1Cond or lowRmv1)
etc2sb = etsb and (rmv2Cond or lowRmv2)
etc3sb = etsb and (rmv3Cond or lowRmv3)

iDivergence = input.int(50, title = 'Min. obv-price div. pct', minval = 0, maxval = 100, group = 'Obv-price divergence sb')
obvMa = ta.sma(ta.obv, 100)
priceMa = ta.sma(close, 100)
risingObv = obvMa > obvMa[1]
risingPrice = priceMa > priceMa[1]
divergence = risingObv and not risingPrice
divergenceSum = math.sum(divergence ? 1 : 0, 100)
obvDiv = divergenceSum > iDivergence and strongBreakout

iIncludeEarnings = input.bool(defval = true, title = 'Include earnings', group = 'Minevrini trend /contraction/ strong breakout')
iBetterEarningsRate = input.float(defval = 0.5, title = 'Min. earnings change rate', minval = 0.0, maxval = 1.0, step = 0.1, group = 'Minevrini trend /contraction/ strong breakout')

doIncludeEarnings = sigMod >= 1 ? false : iIncludeEarnings
bool minevriniTrend = drelib.isMinevriniBullishTrend(true, doIncludeEarnings, iBetterEarningsRate)

minevriniTrendCond = ta.barssince(minevriniTrend) < 63

mtsb = minevriniTrendCond and strongBreakout
mtc1sb = minevriniTrendCond and (rmv1Cond or lowRmv1) and strongBreakout
mtc2sb = minevriniTrendCond and (rmv2Cond or lowRmv2) and strongBreakout
mtc3sb = minevriniTrendCond and (rmv3Cond or lowRmv3) and strongBreakout

[rmv1SuccRate, rmv1SuccCount, rmv1SigCount, rmv1AvgReturn, rmv1SuccRateOk, rmv1SuccRateStr, rmv1Score] = dreperf.successRateAllTimeStatsFull(bullishRmv1, bullishRmv1 ? low : na, iMinSigSuccessRate, adrCond)
[rmv2SuccRate, rmv2SuccCount, rmv2SigCount, rmv2AvgReturn, rmv2SuccRateOk, rmv2SuccRateStr, rmv2Score] = dreperf.successRateAllTimeStatsFull(bullishRmv2, bullishRmv2 ? low : na, iMinSigSuccessRate, adrCond)
[rmv3SuccRate, rmv3SuccCount, rmv3SigCount, rmv3AvgReturn, rmv3SuccRateOk, rmv3SuccRateStr, rmv3Score] = dreperf.successRateAllTimeStatsFull(bullishRmv3, bullishRmv3 ? low : na, iMinSigSuccessRate, adrCond)
[rmv1PbSuccRate, rmv1PbSuccCount, rmv1PbSigCount, rmv1PbAvgReturn, rmv1PbSuccRateOk, rmv1PbSuccRateStr, rmv1PbScore] = dreperf.successRateAllTimeStatsFull(bullishRmv1Pb, bullishRmv1Pb ? low : na, iMinSigSuccessRate, adrCond)
[rmv2PbSuccRate, rmv2PbSuccCount, rmv2PbSigCount, rmv2PbAvgReturn, rmv2PbSuccRateOk, rmv2PbSuccRateStr, rmv2PbScore] = dreperf.successRateAllTimeStatsFull(bullishRmv2Pb, bullishRmv2Pb ? low : na, iMinSigSuccessRate, adrCond)
[rmv3PbSuccRate, rmv3PbSuccCount, rmv3PbSigCount, rmv3PbAvgReturn, rmv3PbSuccRateOk, rmv3PbSuccRateStr, rmv3PbScore] = dreperf.successRateAllTimeStatsFull(bullishRmv3Pb, bullishRmv3Pb ? low : na, iMinSigSuccessRate, adrCond)
[etc1sbSuccRate, etc1sbSuccCount, etc1sbSigCount, etc1sbAvgReturn, etc1sbSuccRateOk, etc1sbSuccRateStr, etc1sbScore] = dreperf.successRateAllTimeStatsFull(etc1sb, etc1sb ? low : na, iMinSigSuccessRate, adrCond)
[etc2sbSuccRate, etc2sbSuccCount, etc2sbSigCount, etc2sbAvgReturn, etc2sbSuccRateOk, etc2sbSuccRateStr, etc2sbScore] = dreperf.successRateAllTimeStatsFull(etc2sb, etc2sb ? low : na, iMinSigSuccessRate, adrCond)
[etc3sbSuccRate, etc3sbSuccCount, etc3sbSigCount, etc3sbAvgReturn, etc3sbSuccRateOk, etc3sbSuccRateStr, etc3sbScore] = dreperf.successRateAllTimeStatsFull(etc3sb, etc3sb ? low : na, iMinSigSuccessRate, adrCond)
[obvDivSuccRate, obvDivSuccCount, obvDivSigCount, obvDivAvgReturn, obvDivSuccRateOk, obvDivSuccRateStr, obvDivScore] = dreperf.successRateAllTimeStatsFull(obvDiv, obvDiv ? low : na, iMinSigSuccessRate, adrCond)
[mtsbSuccRate, mtsbSuccCount, mtsbSigCount, mtsbAvgReturn, mtsbSuccRateOk, mtsbSuccRateStr, mtsbScore] = dreperf.successRateAllTimeStatsFull(mtsb, mtsb ? low : na, iMinSigSuccessRate, adrCond)
[mtc1sbSuccRate, mtc1sbSuccCount, mtc1sbSigCount, mtc1sbAvgReturn, mtc1sbSuccRateOk, mtc1sbSuccRateStr, mtc1sbScore] = dreperf.successRateAllTimeStatsFull(mtc1sb, mtc1sb ? low : na, iMinSigSuccessRate, adrCond)
[mtc2sbSuccRate, mtc2sbSuccCount, mtc2sbSigCount, mtc2sbAvgReturn, mtc2sbSuccRateOk, mtc2sbSuccRateStr, mtc2sbScore] = dreperf.successRateAllTimeStatsFull(mtc2sb, mtc2sb ? low : na, iMinSigSuccessRate, adrCond)
[mtc3sbSuccRate, mtc3sbSuccCount, mtc3sbSigCount, mtc3sbAvgReturn, mtc3sbSuccRateOk, mtc3sbSuccRateStr, mtc3sbScore] = dreperf.successRateAllTimeStatsFull(mtc3sb, mtc3sb ? low : na, iMinSigSuccessRate, adrCond)
