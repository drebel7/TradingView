//@version=6
indicator("early trend strong breakout", overlay = true)

import drebel/DRELib/35 as drelib
import drebel/DREPerf/17 as dreperf

//common filters inputs
iGeneralFilterCond = input.bool(defval = true, title = 'Filter ON', group = 'General filter for signal')
iMinTurnover = input.float(defval = 0.20, title = 'Min. Turnover[M]', minval = 0.2, group = 'General filter for signal')
iMinAdr = input.float(defval = 3, title = 'Min. ADR[%]', minval = 0, step = 0.2, group = 'General filter for signal')
iMinPrice = input.float(defval = 1.0, title = 'Min.-max. price', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxPrice = input.float(defval = 1750.0, title = '-', minval = 0, inline = 'min max price', group = 'General filter for signal')
iMaxDistFromLoDPct = input.int(defval = 67, title = 'Max. distance from LoD [%]', minval = 0, maxval = 300, step = 5, group = 'General filter for signal')
iMinBenchmarkNotBearishSince = input.int(defval = 31, title = 'Min. days from benchmark bearish', minval = 0, maxval = 63, group = 'General filter for signal')

//sig success rate filter
iShowEt3sbSig = input.bool(defval = true, title = 'Show et3sb', group = 'Signals')
iShowEt4sbSig = input.bool(defval = true, title = 'Show et4sb', group = 'Signals')
iShowRmvSig = input.bool(defval = true, title = 'Show rmv', group = 'Signals')
iShowEtcsbSig = input.bool(defval = true, title = 'Show etcsb', group = 'Signals')
iShowObvDivSig = input.bool(defval = true, title = 'Show Obv-Div-sb', group = 'Signals')
iMinSigSuccessRate = input.float(defval = 39.0, title = 'Min. sig success rate pct.', minval = 0.0, maxval = 100.0, step = 1.0, group = 'Signals')
iMinSigReturnAdrMultiple = input.float(defval = 0.85, title = 'Min. avg. return by ADR multiple', minval = -1.0, maxval = 5.0, step = 0.05, group = 'Signals')

//common filters calculations 
int sigMod = drelib.getWideIndexAndSectorSignalMod(iMinBenchmarkNotBearishSince)
bool generalFilter = drelib.generalFilter(iMinTurnover, iMinAdr, iMinPrice, iMaxPrice, 30, iMaxDistFromLoDPct, sigMod)
bool gFilter = generalFilter and iGeneralFilterCond or not iGeneralFilterCond

benchIdx = drelib.getDefaultBenchmarkClose()
rs = close / benchIdx
rsMa = ta.sma(rs, 100)
rsCondRaw = rs > rsMa
rsCond_10 = rsCondRaw ? 1 : 0

obvCondRaw = drelib.isAboveObvSmaStddev(100, 0.35)
obvCond_10 = obvCondRaw ? 1 : 0

iMinNonBrokenLows = input.int(defval = 10, title = 'Min. non-broken Lows', minval = 0, step = 1, maxval = 40, group = 'Other')

[nonBrokenLowsPct, nonBrokenHighsPct, nonBrokenExtremesPct, lowsHighsRatio] = drelib.nonBrokenExtremesPct(100)
nbeCondRaw = nonBrokenLowsPct > nonBrokenHighsPct and nonBrokenExtremesPct >= iMinNonBrokenLows
nbeCond_10 = nbeCondRaw ? 1 : 0

iBreakoutLookback = input.int(defval = 7, title = 'Min. breakout lookback', minval = 1, maxval = 21, group = 'Breakout params')
iBreakoutMultiples = input.bool(defval = true, title = 'Use also multiples of lookback', group = 'Breakout params')
iInitialMinVolumePct = input.int(defval = 600, title = 'Initial min. relative volume Pct.', minval = 100, group = 'Breakout params')
iDayEndMinVolumePct = input.int(defval = 125, title = 'Day end min. relative volume Pct.', minval = 100, group = 'Breakout params')

sigMod += nonBrokenExtremesPct <= 14 ? -1 : 0
strongBreakout = drelib.strongBreakout(iBreakoutLookback, iBreakoutMultiples, iInitialMinVolumePct, iDayEndMinVolumePct, sigMod, true)

aboveIchimokuCloudCondRaw = drelib.isAboveIchimokuCloud()
aboveIchimokuCloudCond_10 = aboveIchimokuCloudCondRaw ? 1 : 0

totalCond = 4
etCondSum = rsCond_10 + obvCond_10 + aboveIchimokuCloudCond_10 + nbeCond_10
et3 = etCondSum == totalCond - 1
et4 = etCondSum == totalCond
et3sb = et3 and strongBreakout
et4sb = et4 and strongBreakout
etsb = et3sb or et4sb

//additionali signal if there is an OpeningRangeBrakeout the next day after main signal
orb = drelib.openingRangeBreakout(5, true, true)

//pct time since session beggining 
dttTime = drelib.dttTimePct()

//rmv sig (like in "DRE parabolic top" v444)
iRmvLookbackPeriod = input.int(30, minval = 1, title = 'Lookback Period', group = 'rmv', display = display.none)
iRmvShortLookback1 = input.int(13, minval = 1, title = 'Short-term ATR Lookback 1', group = 'rmv', display = display.none)
iRmvShortLookback2 = input.int(8, minval = 1, title = 'Short-term ATR Lookback 2', group = 'rmv', display = display.none)
iRmvshortLookback3 = input.int(5, minval = 1, title = 'Short-term ATR Lookback 3', group = 'rmv', display = display.none)
iRmvLowLevel = input.int(25, title = 'Low Level Threshold', group = 'rmv', display = display.none)

rmv = drelib.rmv(iRmvLookbackPeriod, iRmvShortLookback1, iRmvShortLookback2, iRmvshortLookback3)
lowRmv = rmv <= iRmvLowLevel
highRmv = rmv >= 100 - iRmvLowLevel / 2
shortMa = ta.sma(close, 50)
gVShortMa1 = ta.ema(close, 10)
gVolMa = ta.sma(volume, 30)

rmvCond = lowRmv[1] and not lowRmv or highRmv and lowRmv[2] or rmv == 100 and lowRmv[3]
bullishRmv = rmvCond and close > close[2] and volume > gVolMa and close > gVShortMa1 and close > shortMa and dttTime > 0.85

etcsb = etsb and (rmvCond or lowRmv)

iDivergence = input.int(50, title = 'Min. obv-price div. pct', minval = 0, maxval = 100, group = 'Obv-price divergence sb')
obvMa = ta.sma(ta.obv, 100)
priceMa = ta.sma(close, 100)
risingObv = obvMa > obvMa[1]
risingPrice = priceMa > priceMa[1]
divergence = risingObv and not risingPrice
divergenceSum = math.sum(divergence ? 1 : 0, 100)
obvDiv = divergenceSum > iDivergence and strongBreakout

//check performance
iShowSigPerfPlots = input.bool(defval = false, title = 'show plots', group = 'Sig. performnce check')
iShowSigPerfTable = input.bool(defval = false, title = 'show table', group = 'Sig. performnce check')

//sig stats calculation
adrPct = 100.0 * (ta.sma(high / low, 100) - 1)
adrCond = iMinSigReturnAdrMultiple * adrPct

[et3sbSuccRate, et3sbSuccRateOk, et3sbSuccRateStr] = dreperf.successRateAllTimeStatsMin(et3sb, et3sb ? low : na, iMinSigSuccessRate, adrCond)
[et4sbSuccRate, et4sbSuccRateOk, et4sbSuccRateStr] = dreperf.successRateAllTimeStatsMin(et4sb, et4sb ? low : na, iMinSigSuccessRate, adrCond)
[rmvSuccRate, rmvSuccRateOk, rmvSuccRateStr] = dreperf.successRateAllTimeStatsMin(bullishRmv, bullishRmv ? low : na, iMinSigSuccessRate, adrCond)
[etcsbSuccRate, etcsbSuccRateOk, etcsbSuccRateStr] = dreperf.successRateAllTimeStatsMin(etcsb, etcsb ? low : na, iMinSigSuccessRate, adrCond)
[obvDivSuccRate, obvDivSuccRateOk, obvDivSuccRateStr] = dreperf.successRateAllTimeStatsMin(obvDiv, obvDiv ? low : na, iMinSigSuccessRate, adrCond)

//plot sig performance
et3sbSigColor = color.new(color.blue, 0)
et4sbSigColor = color.new(color.lime, 0)
rmvSigColor = color.new(color.maroon, 0)
etcsbSigColor = color.new(color.fuchsia, 0)
obvDivSigColor = color.new(color.navy, 0)

plot(series = iShowSigPerfPlots ? et3sbSuccRate : na, title = 'et3sb', color = et3sbSigColor, linestyle = plot.linestyle_solid)
plot(series = iShowSigPerfPlots ? et4sbSuccRate : na, title = 'et4sb', color = et4sbSigColor, linestyle = plot.linestyle_solid)
plot(series = iShowSigPerfPlots ? rmvSuccRate : na, title = 'rmv', color = rmvSigColor, linestyle = plot.linestyle_solid)
plot(series = iShowSigPerfPlots ? etcsbSuccRate : na, title = 'etcsb', color = etcsbSigColor, linestyle = plot.linestyle_solid)
plot(series = iShowSigPerfPlots ? obvDivSuccRate : na, title = 'obvDiv', color = obvDivSigColor, linestyle = plot.linestyle_solid)

//print sig performance table
if barstate.islast and iShowSigPerfTable
    var tbl = table.new(position = position.top_right, columns = 5, rows = 6)
    table.cell(table_id = tbl, column = 0, row = 0, text = "sig")
    table.cell(table_id = tbl, column = 1, row = 0, text = "perf")

    table.cell(table_id = tbl, column = 0, row = 1, text = "et3sb", text_color = et3sbSigColor)
    table.cell(table_id = tbl, column = 1, row = 1, text = et3sbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 2, text = "et4sb", text_color = et4sbSigColor)
    table.cell(table_id = tbl, column = 1, row = 2, text = et4sbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 3, text = "rmv", text_color = rmvSigColor)
    table.cell(table_id = tbl, column = 1, row = 3, text = rmvSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 4, text = "etcsb", text_color = etcsbSigColor)
    table.cell(table_id = tbl, column = 1, row = 4, text = etcsbSuccRateStr)
    table.cell(table_id = tbl, column = 0, row = 5, text = "obvDiv", text_color = obvDivSigColor)
    table.cell(table_id = tbl, column = 1, row = 5, text = obvDivSuccRateStr)

//plot signals on price chart
plotshape(iShowEt3sbSig and et3sb, title = 'et3sb', location = location.belowbar, color = et3sbSigColor, style = shape.triangleup, text = 'et3sb', textcolor = et3sbSigColor)
plotshape(iShowEt4sbSig and et4sb, title = 'et4sb', location = location.belowbar, color = et4sbSigColor, style = shape.triangleup, text = 'et4sb', textcolor = et4sbSigColor)
plotshape(iShowRmvSig and bullishRmv, title = 'rmv', location = location.belowbar, color = rmvSigColor, style = shape.triangleup, text = 'rmv', textcolor = rmvSigColor)
plotshape(iShowEtcsbSig and etcsb, title = 'etcsb', location = location.belowbar, color = etcsbSigColor, style = shape.triangleup, text = 'etcsb', textcolor = etcsbSigColor)
plotshape(iShowObvDivSig and obvDiv, title = 'obvDiv', location = location.belowbar, color = obvDivSigColor, style = shape.triangleup, text = 'obvDiv', textcolor = obvDivSigColor)

//issuing alerts
et3sbSig = et3sb and et3sbSuccRateOk
et3sbSig_p1D = et3sbSig[1] and orb
et4sbSig = et4sb and et4sbSuccRateOk
et4sbSig_p1D = et4sbSig[1] and orb
bullishRmvSig = bullishRmv and rmvSuccRateOk
bullishRmvSig_p1D = bullishRmvSig[1] and orb
etcsbSig = etcsb and etcsbSuccRateOk
etcsbSig_p1D = etcsbSig[1] and orb
obvDivSig = obvDiv and obvDivSuccRateOk
obvDivSig_p1D = obvDivSig[1] and orb

alertcondition(gFilter and (et3sbSig or et3sbSig_p1D), title = 'bullish ET3SB', message = 'ET3SB')
alertcondition(gFilter and (et4sbSig or et4sbSig_p1D), title = 'bullish ET4SB', message = 'ET4SB')
alertcondition(gFilter and (bullishRmvSig or bullishRmvSig_p1D), title = 'bullish RMV', message = 'RMV')
alertcondition(gFilter and (etcsbSig or etcsbSig_p1D), title = 'bullish ETCSB', message = 'ETCSB')
alertcondition(gFilter and (obvDivSig or obvDivSig_p1D), title = 'bullish obvDiv', message = 'obvDiv')
alertcondition(gFilter and (et3sbSig or et3sbSig_p1D or et4sbSig or et4sbSig_p1D or bullishRmvSig or bullishRmvSig_p1D or etcsbSig or etcsbSig_p1D or obvDivSig or obvDivSig_p1D), title = 'best LONG', message = 'best LONG')
