strategy("DRE trippleSig", overlay=false, initial_capital=100, default_qty_type=strategy.percent_of_equity, default_qty_value=100, commission_type=strategy.commission.percent, commission_value=0.29, margin_long=10000, margin_short=10000)

ma(source, length, type) =>
    switch type
        "SMA" => ta.sma(source, length)
        "EMA" => ta.ema(source, length)
        "SMMA (RMA)" => ta.rma(source, length)
        "WMA" => ta.wma(source, length)
        "VWMA" => ta.vwma(source, length)


//OBV
iObvSigActive = input.bool(title='OBV signal active', defval=true, group='sig: OBV', inline='obv0')
iObvMaType = input.string(title='MA Type', defval='SMMA (RMA)', group='sig: OBV', options=['SMA', 'EMA', 'SMMA (RMA)', 'WMA', 'VWMA'], inline='obv1')
iObvMaLength = input.int(title='Len', defval=100, group='sig: OBV', minval=10, maxval=250, inline='obv1')

var cumVol = 0.
cumVol += nz(volume)
if barstate.islast and cumVol == 0
    runtime.error("No volume is provided by the data vendor.")
obv = ta.cum(math.sign(ta.change(close)) * volume)
obcMa = ma(obv, iObvMaLength, iObvMaType)
//plot(obv, color=#2962FF, title="OnBalanceVolume")
obvSig = obv > obcMa
obvSig2 = iObvSigActive and obvSig or not iObvSigActive
p1 = plot(obvSig2 ? 1 : 0, title='obvSig', color=color.new(color.purple, 0))
fill(plot1=plot(0, color=color.new(color.black, 100)), plot2=p1, color=color.new(color.purple, 80))
if barstate.islast
    label.new(x=last_bar_index, y=0, text="OBV", xloc=xloc.bar_index, yloc=yloc.price, style=label.style_none, textalign=text.align_left, textcolor=color.new(color.purple, 0))


//MA %
iMaPctSigActive = input.bool(title='MA % signal active', defval=true, group='sig: MA %', inline='ma%0')
iMaPctMaType = input.string(title='MA type', group='sig: MA %', defval='SMA', options=['SMA', 'EMA', 'SMMA (RMA)', 'WMA', 'VWMA'], inline='ma%1')
iMaPctMaLength = input.int(title='Len', defval=200, group='sig: MA %', inline='ma%1')

maPctMa = ma(close, iMaPctMaLength, iMaPctMaType)
maPctDiff = (close - maPctMa) / maPctMa * 100
maPctSig = maPctDiff > 0
maPctSig2 = iMaPctSigActive and maPctSig or not iMaPctSigActive
p2 = plot(maPctSig2 ? 3 : 2, title='maPctSig', color=color.new(color.orange, 0))
fill(plot1=plot(2, color=color.new(color.black, 100)), plot2=p2, color=color.new(color.orange, 80))
if barstate.islast
    label.new(x=last_bar_index, y=2, text="MA", xloc=xloc.bar_index, yloc=yloc.price, style=label.style_none, textalign=text.align_left, textcolor=color.new(color.orange, 0))


//Ichimoku cloud
donchian(len) => math.avg(ta.lowest(len), ta.highest(len))

iIchimokuSigActive = input.bool(title='Ichimoku signal active', defval=true, group='sig: Ichi', inline='ichi0')
conversionPeriods = 9
basePeriods = 26
laggingSpan2Periods = 52
displacement = 26
conversionLine = donchian(conversionPeriods)
baseLine = donchian(basePeriods)
leadLine1 = math.avg(conversionLine, baseLine)
leadLine2 = donchian(laggingSpan2Periods)
//plot(leadLine1 > leadLine2 ? leadLine1 : leadLine2, offset = displacement - 1, title = "Kumo Cloud Upper Line", display = display.none) 
plot(leadLine1 < leadLine2 ? leadLine1 : leadLine2, offset = displacement - 1, title = "Kumo Cloud Lower Line", display = display.none) 
ichimokuSigL = close > math.max(leadLine1[displacement], leadLine2[displacement])
ichimokuSigS = close < math.min(leadLine1[displacement], leadLine2[displacement])
ichimokuSig = ta.barssince(ichimokuSigL) < ta.barssince(ichimokuSigS)
ichimokuSig2 = iIchimokuSigActive and ichimokuSig or not iIchimokuSigActive
pIchi = plot(ichimokuSig2 ? 5 : 4, title='ichimokuSig', color=color.new(color.lime, 0))
fill(plot1=plot(4, color=color.new(color.lime, 100)), plot2=pIchi, color=color.new(color.lime, 80))
if barstate.islast
    label.new(x=last_bar_index, y=4, text="Ichimoku", xloc=xloc.bar_index, yloc=yloc.price, style=label.style_none, textalign=text.align_left, textcolor=color.new(color.lime, 0))



//Mansfield Relative Strength
iMrsSigActive = input.bool(title='MRS signal active', defval=true, group='sig: MRS')
iMrsIndex = input.symbol(title='Index', defval='GPW:FW201!', group='sig: MRS')
iMrsMaType = input.string(title='MA type', group='sig: MRS', defval='WMA', options=['SMA', 'EMA', 'SMMA (RMA)', 'WMA', 'VWMA'], inline='mrs1')
iMrsMaLen = input.int(title='Len', defval=200, group='sig: MRS', inline='mrs1')

calc_mrs_func(x, ind, length, maType) =>
    (x / request.security(ind, timeframe.period, x) / ma(x / request.security(ind, timeframe.period, x), length, maType) - 1) * 10

mrsIndex = calc_mrs_func(close, iMrsIndex, iMrsMaLen, iMrsMaType)
mrsSig = mrsIndex > 0
mrsSig2 = iMrsSigActive and mrsSig or not iMrsSigActive
p4 = plot(mrsSig2 ? 7 : 6, title='mrsSig', color=color.new(color.black, 0))
fill(plot1=plot(6, color=color.new(color.black, 100)), plot2=p4, color=color.new(color.black, 80))
if barstate.islast
    label.new(x=last_bar_index, y=6, text="MRS", xloc=xloc.bar_index, yloc=yloc.price, style=label.style_none, textalign=text.align_left, textcolor=color.new(color.black, 0))



//MA coiling
iMaCoilSigActive = input.bool(title='MA Coil signal active', defval=true, group='MA coil')
iMaCoilMaType = input.string(title='MA type', defval='SMA', group='MA coil', options=['SMA', 'EMA', 'SMMA (RMA)', 'WMA', 'VWMA'])
iMaCoilMaLength1 = input.int(title='MA Lengths', defval=50, group='MA coil', inline='ma coil mas')
iMaCoilMaLength2 = input.int(title='', defval=100, group='MA coil', inline='ma coil mas')
iMaCoilMaLength3 = input.int(title='', defval=200, group='MA coil', inline='ma coil mas')
iMaCoilPct = input.float(title='MA Coil %', defval=4.5, group='MA coil' , minval = 0, maxval = 10, step=0.2)

ma0 = ma(close, 20, iMaCoilMaType)
ma1 = ma(close, iMaCoilMaLength1, iMaCoilMaType)
ma2 = ma(close, iMaCoilMaLength2, iMaCoilMaType)
ma3 = ma(close, iMaCoilMaLength3, iMaCoilMaType)
min_ma123 = math.min(ma1, ma2, ma3)

diff1_pct = (close - ma1) / ma1 * 100
diff2_pct = (close - ma2) / ma2 * 100
diff3_pct = (close - ma3) / ma3 * 100
diff123_pct = (math.abs(diff1_pct) + math.abs(diff2_pct) + math.abs(diff3_pct)) / 3

//consecutiveCoiledDays = ta.barssince(diff123_pct > iMaCoilPct)
//plot(consecutiveCoiledDays, color=color.new(color.black, 0))
maCoilSituations = math.sum(diff123_pct <= iMaCoilPct ? 1 : 0, 50)
maCoilSig = maCoilSituations >= 1 and diff123_pct <= 2 * iMaCoilPct
maCoilSig2 = iMaCoilSigActive and maCoilSig or not iMaCoilSigActive
p3 = plot(maCoilSig2 ? 9 : 8, title='maCoilSig', color=color.new(color.blue, 0))
fill(plot1=plot(8, color=color.new(color.black, 100)), plot2=p3, color=color.new(color.blue, 80))
maCoil10Pct = ta.percentile_nearest_rank(diff123_pct, math.min(2000, math.max(bar_index, iObvMaLength, iMaPctMaLength)), 10)
if barstate.islast
    label.new(x=last_bar_index, y=8, text="MA coil " + str.tostring(maCoil10Pct, '#.#'), xloc=xloc.bar_index, yloc=yloc.price, style=label.style_none, textalign=text.align_right, textcolor=color.new(color.blue, 0))



//position close parameters
iBullBg = input.bool(title='Bullish background', defval=true, group='general')
iBaerBg = input.bool(title='Bearish background', defval=true, group='general')
iStopLossPct = input.float(title='Stop loss % (from entry)', defval=6 , minval = 1, maxval = 12, step=1, group='general')


initialBuyCondition = strategy.opentrades == 0
initialSellCondition = strategy.opentrades > 0
entryId = "trippleSig"

isEnoughData = bar_index > math.max(iObvMaLength, iMaPctMaLength)
bgcolor(isEnoughData and iBullBg and initialBuyCondition and obvSig2 and maPctSig2 and ichimokuSig2 and mrsSig2 ? color.new(color.green, 70) : na)
bgcolor(isEnoughData and iBaerBg and initialBuyCondition and not obvSig2 and not maPctSig2 and not ichimokuSig2 and not mrsSig2 ? color.new(color.red, 80) : na)
//buy
if (initialBuyCondition and obvSig2 and maPctSig2 and ichimokuSig2 and mrsSig2)
    strategy.entry(id=entryId, direction=strategy.long)

currEntryPrice = strategy.opentrades.entry_price(strategy.opentrades - 1)
hardStopLossLvl = currEntryPrice * (1 - iStopLossPct / 100)
hardStopLossSig = close <= hardStopLossLvl
movingStopLossLvl = close > 3 * currEntryPrice ? ma0 : close > 2 * currEntryPrice ? ma1 : close > 1.5 * currEntryPrice ? ma2 : min_ma123 * (1 - iStopLossPct / 2 / 100)
movingStopLossSig = close <= movingStopLossLvl

//sell
if (initialSellCondition and (hardStopLossSig or movingStopLossSig))
    txt = hardStopLossSig ? 'hard sl' : 'moving sl'
    strategy.close(id=entryId, comment=txt)
