//@version=6
indicator("Non-Broken lows and highs", overlay=false)

lookback = input.int(defval = 100, title = "Lookback", minval = 10, maxval = 500)
plottHighLvl = input.int(defval = 24, title = "High Level", minval = 0, maxval = 100)
ratioThreshold = input.float(defval = 3.0, title = "Ratio", minval = 1.0, maxval = 10.0, step = 0.25)

int minCount = 0
int maxCount = 0
float lowSoFar = low
float highSoFar = high
for int i = 1 to lookback
    if (low[i] < lowSoFar)
        minCount += 1
        lowSoFar := low[i]
    if (high[i] > highSoFar)
        maxCount += 1
        highSoFar := high[i]

float lowestSumPct = 100.0 * minCount / lookback
float highestSumPct = 100.0 * maxCount / lookback
float allPct = lowestSumPct + highestSumPct
float ratio = 1.0 * minCount / math.max(maxCount, 1)

plot(lowestSumPct, "Non-broken lows", color=color.green, linewidth=2)
plot(highestSumPct, "Non-broken highs", color=color.red, linewidth=2)
plot(allPct, "Non-broken all", color=color.black, linewidth=1)
bgcolor(ratio < 1 / ratioThreshold and allPct > plottHighLvl ? color.new(color.red, 90) : ratio > ratioThreshold and allPct > plottHighLvl ? color.new(color.green, 90) : na)
